335def test_scan_records_errors_without_aborting(make_repo, tmp_path, monkeypatch):
336 make_repo({"a.py": "x = 1\n"}, name="repo-a")
337 make_repo({"b.py": "x = 1\n"}, name="repo-b")
338
339 import trinkets.repostats.multi as multi_module
340
341 real_analyse = multi_module.analyse_repository
342
343 def flaky_analyse(repo_path, **kwargs):
344 if repo_path.name == "repo-b":
345 raise multi_module.GitError("simulated failure")
346 return real_analyse(repo_path, **kwargs)
347
348 monkeypatch.setattr(multi_module, "analyse_repository", flaky_analyse)
349
350 entries, output_dir = scan(tmp_path, tmp_path / "report")
351
352 by_slug = {e.slug: e for e in entries}
353 assert by_slug["repo-a"].report is not None
354 assert by_slug["repo-b"].error == "simulated failure"
355 assert (output_dir / "repo-a" / "report.html").exists()
356 assert not (output_dir / "repo-b").exists()
357 assert (output_dir / "index.html").exists()
