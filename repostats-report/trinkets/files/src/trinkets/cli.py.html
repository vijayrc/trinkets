src/trinkets/cli.py

1"""Top-level dispatcher for the trinkets utility collection.
2
3Each utility lives in its own subpackage and exposes a ``register(subparsers)``
4function plus a ``run(args)`` entry point. Adding a new tool means adding one
5module here and one line to ``UTILITIES``.
6"""
7
8from __future__ import annotations
9
10import argparse
11import sys
12from collections.abc import Sequence
13
14from trinkets import __version__
15
16# name -> (module path, one-line help). Imported lazily so a broken or
17# heavyweight utility never slows down or breaks the others.
18UTILITIES: dict[str, tuple[str, str]] = {
19 "repostats": (
20 "trinkets.repostats.cli",
21 "Analyse a git repository and report languages, build, deps, flow and test stats.",
22 ),
23}
24
25
26def build_parser() -> argparse.ArgumentParser:
27 parser = argparse.ArgumentParser(
28 prog="trinkets",
29 description="A collection of small, self-contained developer utilities.",
30 )
31 parser.add_argument("--version", action="version", version=f"trinkets {__version__}")
32
33 subparsers = parser.add_subparsers(dest="utility", metavar="<utility>")
34 for name, (module_path, help_text) in UTILITIES.items():
35 sub = subparsers.add_parser(name, help=help_text, add_help=False)
36 sub.set_defaults(_module=module_path)
37 return parser
38
39
40def main(argv: Sequence[str] | None = None) -> int:
41 argv = list(sys.argv[1:] if argv is None else argv)
42
43 # Dispatch on the first token so each utility owns its full argument
44 # namespace (including -h) rather than fighting the parent parser.
45 if argv and argv[0] in UTILITIES:
46 module_path, _ = UTILITIES[argv[0]]
47 module = __import__(module_path, fromlist=["main"])
48 return int(module.main(argv[1:]))
49
50 parser = build_parser()
51 if not argv or argv[0] in {"-h", "--help"}:
52 parser.print_help()
53 return 0
54 parser.parse_args(argv) # handles --version and reports unknown utilities
55 parser.print_help()
56 return 1
57
58
59if __name__ == "__main__":
60 raise SystemExit(main())
