16EPILOG = """\
17examples:
18 repostats analyse the current directory
19 repostats ~/code/myapp analyse another repository
20 repostats -f json -o report.json emit machine-readable output
21 repostats -f terminal print a colourised report to the terminal
22 repostats --run-coverage execute the repo's pytest suite to measure coverage
23 repostats scan ~/code -o report/ analyse every repo under a folder into one HTML site
24
25note:
26 --run-coverage executes the analysed repository's test suite. Only use it on
27 code you trust. Without it, coverage is read from existing reports only.
28"""
29
30SCAN_EPILOG = """\
31examples:
32 repostats scan ~/code analyse every repo under ~/code
33 repostats scan ~/code -o site/ write the HTML report to site/ instead
34
35Produces an index.html linking to one report.html per repository, and each
36report links out to the specific source files it mentions (manifests, largest
37files, API endpoint locations, ...).
38"""

41def build_parser() -> argparse.ArgumentParser:
42 parser = argparse.ArgumentParser(
43 prog="repostats",
44 description="Analyse a git repository: languages, build, purpose, contributors, "
45 "flow, dependencies, size and tests.",
46 epilog=EPILOG,
47 formatter_class=argparse.RawDescriptionHelpFormatter,
48 )
49 parser.add_argument(
50 "path", nargs="?", default=".", help="repository to analyse (default: current directory)"
51 )
52 parser.add_argument(
53 "-f", "--format", choices=("markdown", "json", "terminal"), default="markdown",
54 help="output format (default: markdown); 'terminal' renders the same report "
55 "with ANSI colour for reading on screen",
56 )
57 parser.add_argument(
58 "-o", "--output", metavar="FILE", help="write to FILE instead of stdout",
59 )
60 parser.add_argument(
61 "--color", choices=("auto", "always", "never"), default="auto",
62 help="colour control for --format terminal (default: auto)",
63 )
64 parser.add_argument(
65 "--max-commits", type=int, default=None, metavar="N",
66 help="only inspect the most recent N commits (faster on huge histories)",
67 )
68 parser.add_argument(
69 "--max-file-size", type=int, default=2_000_000, metavar="BYTES",
70 help="skip files larger than this (default: 2000000)",
71 )
72 parser.add_argument(
73 "--run-coverage", action="store_true",
74 help="execute the repository's pytest suite to measure coverage (runs untrusted code)",
75 )
76 parser.add_argument("--version", action="version", version=f"repostats {__version__}")
77 return parser

80def build_scan_parser() -> argparse.ArgumentParser:
81 parser = argparse.ArgumentParser(
82 prog="repostats scan",
83 description="Analyse every git repository under a folder and publish one consolidated "
84 "HTML report: an index page linking to a per-repo report, each of which "
85 "links out to the source files it references.",
86 epilog=SCAN_EPILOG,
87 formatter_class=argparse.RawDescriptionHelpFormatter,
88 )
89 parser.add_argument(
90 "path", nargs="?", default=".",
91 help="folder to search for repositories (default: current directory)",
92 )
93 parser.add_argument(
94 "-o", "--output", metavar="DIR", default="repostats-report",
95 help="directory to write the HTML report tree to (default: ./repostats-report)",
96 )
97 parser.add_argument(
98 "--max-depth", type=int, default=6, metavar="N",
99 help="how many directory levels to search for repositories (default: 6)",
100 )
101 parser.add_argument(
102 "--max-commits", type=int, default=None, metavar="N",
103 help="only inspect the most recent N commits per repository",
104 )
105 parser.add_argument(
106 "--max-file-size", type=int, default=2_000_000, metavar="BYTES",
107 help="skip files larger than this (default: 2000000)",
108 )
109 parser.add_argument(
110 "--run-coverage", action="store_true",
111 help="execute each repository's test suite to measure coverage (runs untrusted code)",
112 )
113 return parser

116def _scan_main(argv: Sequence[str] | None) -> int:
117 parser = build_scan_parser()
118 args = parser.parse_args(argv)
119
120 root = Path(args.path).expanduser()
121 if not root.is_dir():
122 print(f"repostats scan: {root} is not a directory", file=sys.stderr)
123 return 2
124
125 def log(display_path: str) -> None:
126 print(f"analysing {display_path} ...", file=sys.stderr)
127
128 try:
129 entries, output_dir = scan(
130 root,
131 Path(args.output).expanduser(),
132 max_depth=args.max_depth,
133 max_commits=args.max_commits,
134 max_file_bytes=args.max_file_size,
135 run_coverage=args.run_coverage,
136 on_progress=log,
137 )
138 except KeyboardInterrupt:
139 print("repostats scan: interrupted", file=sys.stderr)
140 return 130
141
142 if not entries:
143 print(f"repostats scan: no git repositories found under {root}", file=sys.stderr)
144 return 1
145
146 ok = sum(1 for e in entries if e.report is not None)
147 failed = len(entries) - ok
148 print(f"Wrote {ok} report(s) ({failed} failed) to {output_dir / 'index.html'}", file=sys.stderr)
149 return 0
150
151
152def main(argv: Sequence[str] | None = None) -> int:
153 argv = list(sys.argv[1:] if argv is None else argv)
154 if argv and argv[0] == "scan":
155 return _scan_main(argv[1:])
156
157 parser = build_parser()
158 args = parser.parse_args(argv)
159
160 repo_path = Path(args.path).expanduser()
161 if not repo_path.exists():
162 print(f"repostats: {repo_path} does not exist", file=sys.stderr)
163 return 2
164
165 try:
166 report = analyse_repository(
167 repo_path,
168 max_commits=args.max_commits,
169 run_coverage=args.run_coverage,
170 max_file_bytes=args.max_file_size,
171 )
172 except NotADirectoryError as exc:
173 print(f"repostats: {exc}", file=sys.stderr)
174 return 2
175 except GitError as exc:
176 print(f"repostats: git error: {exc}", file=sys.stderr)
177 return 1
178 except KeyboardInterrupt:
179 print("repostats: interrupted", file=sys.stderr)
180 return 130
181
182 if args.format == "json":
183 rendered = render_json(report)
184 elif args.format == "terminal":
185 use_color = args.color == "always" or (
186 args.color == "auto" and not args.output and sys.stdout.isatty()
187 )
188 rendered = render_terminal(report, color=use_color)
189 else:
190 rendered = render_markdown(report)
191
192 if args.output:
193 output_path = Path(args.output).expanduser()
194 output_path.parent.mkdir(parents=True, exist_ok=True)
195 output_path.write_text(rendered, encoding="utf-8")
196 print(f"Wrote {args.format} report to {output_path}", file=sys.stderr)
197 else:
198 print(rendered)
199 return 0
200
201
202if __name__ == "__main__":
203 raise SystemExit(main())
