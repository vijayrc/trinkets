377def normalise(name: str) -> str:
378 return name.strip().lower().replace("_", "-")
379
380
381def classify_package(name: str) -> tuple[str, str] | None:
382 """Return (category, canonical system) for a dependency name, if known."""
383 key = normalise(name)
384 if key in PACKAGE_SYSTEMS:
385 return PACKAGE_SYSTEMS[key]
386
387 # Java/Go coordinates: try the artifact id or final path segment.
388 for separator in (":", "/"):
389 if separator in key:
390 tail = key.rsplit(separator, 1)[-1]
391 if tail in PACKAGE_SYSTEMS:
392 return PACKAGE_SYSTEMS[tail]
393
394 for fragment, category, system in SUBSTRING_SYSTEMS:
395 if fragment in key:
396 return category, system
397 return None
398
399
400def classify_image(image: str) -> tuple[str, str] | None:
401 """Return (category, system) for a container image reference."""
402 ref = image.strip().lower().split("@", 1)[0]
403 ref = ref.rsplit(":", 1)[0] # drop tag
404 ref = ref.rsplit("/", 1)[-1] # drop registry/namespace
405 for fragment, category, system in DOCKER_IMAGE_SYSTEMS:
406 if fragment in ref:
407 return category, system
408 return None
