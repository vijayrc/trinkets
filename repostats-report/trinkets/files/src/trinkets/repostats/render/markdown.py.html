7CONFIDENCE_BADGE = {
8 Confidence.MEASURED: "measured",
9 Confidence.HIGH: "high confidence",
10 Confidence.MEDIUM: "medium confidence — inferred",
11 Confidence.LOW: "low confidence — treat as a hint",
12}

15def _pct(part: int, whole: int) -> str:
16 return f"{(part / whole * 100):.1f}%" if whole else "—"
17

19def _humanise_bytes(count: int) -> str:
20 size = float(count)
21 for unit in ("B", "KB", "MB", "GB"):
22 if size < 1024 or unit == "GB":
23 return f"{size:.0f} {unit}" if unit == "B" else f"{size:.1f} {unit}"
24 size /= 1024
25 return f"{size:.1f} GB"
26

28def _badge(confidence: Confidence) -> str:
29 return f"*({CONFIDENCE_BADGE.get(confidence, confidence.value)})*"

32def _section_languages(report: RepoReport) -> list[str]:
33 lines = ["## 1. Programming languages", ""]
34 code_languages = [entry for entry in report.languages if entry.is_code]
35 other_languages = [entry for entry in report.languages if not entry.is_code]
36
37 if not code_languages:
38 lines += ["No recognised source code found.", ""]
39 return lines
40
41 total_code = sum(entry.code_lines for entry in code_languages) or 1
42 lines += [
43 "| Language | Files | Code lines | Share | Comments | Blank |",
44 "| --- | ---: | ---: | ---: | ---: | ---: |",
45 ]
46 for entry in code_languages[:15]:
47 lines.append(
48 f"| {entry.language} | {entry.files:,} | {entry.code_lines:,} | "
49 f"{_pct(entry.code_lines, total_code)} | {entry.comment_lines:,} | "
50 f"{entry.blank_lines:,} |"
51 )
52 lines.append("")
53 primary = code_languages[0]
54 lines.append(
55 f"**Primary language:** {primary.language} "
56 f"({_pct(primary.code_lines, total_code)} of source lines)."
57 )
58
59 if other_languages:
60 summary = ", ".join(
61 f"{entry.language} ({entry.files})" for entry in other_languages[:8]
62 )
63 lines += ["", f"**Config / markup files:** {summary}."]
64 lines.append("")
65 return lines

68def _section_build(report: RepoReport) -> list[str]:
69 build = report.build
70 lines = ["## 2. Build tooling and project model", ""]
71
72 rows = [
73 ("Build tools", ", ".join(build.tools) or "none detected"),
74 ("Package managers", ", ".join(build.package_managers) or "none detected"),
75 ("PEP 517 backend", build.backend or "—"),
76 ("Project model", build.model or "unknown"),
77 ("CI systems", ", ".join(build.ci_systems) or "none detected"),
78 ("Containerised", "yes" if build.containerised else "no"),
79 ]
80 lines += ["| Aspect | Finding |", "| --- | --- |"]
81 lines += [f"| {label} | {value} |" for label, value in rows]
82 lines.append("")
83
84 if build.manifests:
85 lines += ["**Manifests found:** " + ", ".join(f"`{m}`" for m in build.manifests[:12]), ""]
86 if build.workspaces:
87 shown = ", ".join(f"`{w}`" for w in build.workspaces[:12])
88 more = f" (+{len(build.workspaces) - 12} more)" if len(build.workspaces) > 12 else ""
89 lines += [f"**Workspace members:** {shown}{more}", ""]
90 return lines

93def _section_purpose(report: RepoReport) -> list[str]:
94 purpose = report.purpose
95 lines = [f"## 3. Purpose and overall logic {_badge(purpose.confidence)}", ""]
96 lines += [purpose.summary, ""]
97
98 if purpose.declared_description:
99 lines += [f"**Declared description:** {purpose.declared_description}", ""]
100 if purpose.readme_excerpt and purpose.readme_excerpt != purpose.declared_description:
101 lines += ["**README says:**", "", f"> {purpose.readme_excerpt}", ""]
102 if purpose.detected_kinds:
103 lines += ["**Detected project kind(s):** " + ", ".join(purpose.detected_kinds), ""]
104 if purpose.frameworks:
105 lines += ["**Frameworks / platforms:** " + ", ".join(purpose.frameworks), ""]
106 if purpose.domain_terms:
107 lines += [
108 "**Recurring domain terms in paths:** "
109 + ", ".join(f"`{term}`" for term in purpose.domain_terms),
110 "",
111 ]
112 if purpose.evidence:
113 lines.append("<details><summary>Evidence</summary>")
114 lines.append("")
115 for item in purpose.evidence:
116 location = f" — `{item.path}`" if item.path else ""
117 lines.append(f"- {item.detail}{location}")
118 lines += ["", "</details>", ""]
119 return lines

122def _section_contributors(report: RepoReport) -> list[str]:
123 info = report.contributors
124 lines = ["## 4. Contributors and commit timeframe", ""]
125
126 if not info.total_commits:
127 lines += ["No commit history available.", ""]
128 return lines
129
130 span = "—"
131 if info.first_commit and info.last_commit:
132 span = f"{info.first_commit} → {info.last_commit}"
133
134 lines += [
135 "| Metric | Value |",
136 "| --- | --- |",
137 f"| Total commits (excl. merges) | {info.total_commits:,} |",
138 f"| Distinct authors | {info.total_authors:,} |",
139 f"| Active span | {span} |",
140 f"| Days with commits | {info.active_days:,} |",
141 f"| Bus factor | {info.bus_factor} |",
142 "",
143 ]
144 if info.bus_factor_note:
145 lines += [f"*{info.bus_factor_note}*", ""]
146
147 lines += [
148 "### Top contributors",
149 "",
150 "| Author | Commits | Share | Insertions | Deletions | Active |",
151 "| --- | ---: | ---: | ---: | ---: | --- |",
152 ]
153 for person in info.top_contributors[:10]:
154 active = f"{person.first_commit or '?'} → {person.last_commit or '?'}"
155 lines.append(
156 f"| {person.name} | {person.commits:,} | "
157 f"{_pct(person.commits, info.total_commits)} | {person.insertions:,} | "
158 f"{person.deletions:,} | {active} |"
159 )
160 lines.append("")
161
162 if info.commits_by_year:
163 peak = max(info.commits_by_year.values()) or 1
164 lines += ["### Commits by year", "", "```"]
165 for year, count in info.commits_by_year.items():
166 bar = "█" * max(1, round(count / peak * 40))
167 lines.append(f"{year} {bar} {count:,}")
168 lines += ["```", ""]
169 return lines

172def _section_flow(report: RepoReport) -> list[str]:
173 flow = report.flow
174 lines = [f"## 5. Flow: entry to persistence {_badge(flow.confidence)}", ""]
175
176 if flow.note:
177 lines += [f"> {flow.note}", ""]
178
179 if flow.mermaid:
180 lines += ["```mermaid", flow.mermaid, "```", ""]
181
182 interesting = [node for node in flow.nodes if node.evidence]
183 if interesting:
184 lines.append("<details><summary>How each node was identified</summary>")
185 lines.append("")
186 for node in interesting:
187 label = node.label.replace("<br/>", " ")
188 lines.append(f"- **{label}**")
189 for item in node.evidence:
190 location = f" — `{item.path}`" if item.path else ""
191 lines.append(f" - {item.detail}{location}")
192 lines += ["", "</details>", ""]
193 return lines

196def _section_dependencies(report: RepoReport) -> list[str]:
197 deps = report.dependencies
198 lines = ["## 6. External dependencies and infrastructure", ""]
199
200 if deps.by_category:
201 lines += ["### Systems this code talks to", "", "| Category | Systems |", "| --- | --- |"]
202 for category, systems in deps.by_category.items():
203 lines.append(f"| {category.title()} | {', '.join(systems)} |")
204 lines.append("")
205 else:
206 lines += ["No databases, caches, queues or external services detected.", ""]
207
208 if deps.infrastructure:
209 lines += ["### Infrastructure evidence", "", "| System | Category | Evidence | Source |",
210 "| --- | --- | --- | --- |"]
211 for dep in deps.infrastructure[:20]:
212 lines.append(
213 f"| {dep.name} | {dep.category} | {dep.detail or '—'} | `{dep.source}` |"
214 )
215 lines.append("")
216
217 lines += [f"**Declared dependencies:** {deps.total_declared:,} across all manifests.", ""]
218
219 notable = [dep for dep in deps.declared if dep.category != "other"]
220 if notable:
221 lines.append("<details><summary>Classified declared dependencies</summary>")
222 lines += ["", "| Package | Version | Category | Implies | Manifest |",
223 "| --- | --- | --- | --- | --- |"]
224 for dep in sorted(notable, key=lambda d: (d.category, d.name))[:60]:
225 lines.append(
226 f"| `{dep.name}` | {dep.version or '—'} | {dep.category} | "
227 f"{dep.detail or '—'} | `{dep.source}` |"
228 )
229 lines += ["", "</details>", ""]
230 return lines

233def _section_code_stats(report: RepoReport) -> list[str]:
234 code = report.code
235 lines = ["## 7. Size and structure", ""]
236
237 lines += [
238 "| Metric | Value |",
239 "| --- | ---: |",
240 f"| Files analysed | {code.analysed_files:,} |",
241 f"| Files skipped (binary, vendored, generated) | {code.skipped_files:,} |",
242 f"| Total lines | {code.total_lines:,} |",
243 f"| Code lines | {code.code_lines:,} |",
244 f"| Comment lines | {code.comment_lines:,} |",
245 f"| Blank lines | {code.blank_lines:,} |",
246 f"| Comment ratio | {_pct(code.comment_lines, code.code_lines + code.comment_lines)} |",
247 f"| Classes / types | {code.classes:,} |",
248 f"| Functions / methods | {code.functions:,} |",
249 f"| HTTP endpoints | {code.api_endpoints:,} |",
250 f"| Repository size (analysed) | {_humanise_bytes(code.bytes)} |",
251 "",
252 ]
253
254 if code.exact_parse_languages or code.heuristic_parse_languages:
255 notes = []
256 if code.exact_parse_languages:
257 notes.append(
258 f"exact AST parsing for {', '.join(code.exact_parse_languages)}"
259 )
260 if code.heuristic_parse_languages:
261 notes.append(
262 f"regex heuristics for {', '.join(code.heuristic_parse_languages)} "
263 "(counts are approximate)"
264 )
265 lines += [f"> Class/function counts use {'; '.join(notes)}.", ""]
266
267 if code.endpoints:
268 lines += ["### API endpoints", "", "| Method | Route | Location | Detected as |",
269 "| --- | --- | --- | --- |"]
270 for endpoint in code.endpoints[:40]:
271 lines.append(
272 f"| `{endpoint.method}` | `{endpoint.route}` | "
273 f"`{endpoint.path}:{endpoint.line}` | {endpoint.framework} |"
274 )
275 if len(code.endpoints) > 40:
276 lines.append(f"| … | *{len(code.endpoints) - 40} more* | | |")
277 lines.append("")
278
279 if code.counts_by_language:
280 lines += ["### Declarations by language", "",
281 "| Language | Classes / types | Functions |", "| --- | ---: | ---: |"]
282 for language, counts in sorted(
283 code.counts_by_language.items(), key=lambda item: -item[1]["functions"]
284 ):
285 lines.append(f"| {language} | {counts['classes']:,} | {counts['functions']:,} |")
286 lines.append("")
287
288 if code.largest_files:
289 lines += ["### Largest source files", "", "| File | Lines |", "| --- | ---: |"]
290 for path, count in code.largest_files:
291 lines.append(f"| `{path}` | {count:,} |")
292 lines.append("")
293 return lines

296def _section_testing(report: RepoReport) -> list[str]:
297 testing = report.testing
298 lines = ["## 8. Testing and coverage", ""]
299
300 coverage = (
301 f"{testing.coverage_percent:.1f}%" if testing.coverage_percent is not None else "not measured"
302 )
303 lines += [
304 "| Metric | Value |",
305 "| --- | --- |",
306 f"| Test frameworks | {', '.join(testing.frameworks) or 'none detected'} |",
307 f"| Coverage tooling configured | {', '.join(testing.coverage_tools_configured) or 'none'} |",
308 f"| Test files | {testing.test_files:,} |",
309 f"| Source files (non-test) | {testing.source_files:,} |",
310 f"| Test-to-source file ratio | {testing.test_to_source_ratio:.2f} |",
311 f"| Test functions / cases | {testing.test_functions:,} |",
312 f"| Lines of test code | {testing.test_lines:,} |",
313 f"| Line coverage | {coverage} |",
314 "",
315 ]
316 if testing.coverage_source:
317 lines += [f"Coverage read from `{testing.coverage_source}`.", ""]
318 if testing.note:
319 lines += [f"> {testing.note}", ""]
320 return lines

323def _header(report: RepoReport) -> list[str]:
324 lines = [
325 f"# Repository report: {report.name}",
326 "",
327 f"`{report.path}`",
328 "",
329 "| | |",
330 "| --- | --- |",
331 f"| Generated | {report.generated_at} |",
332 f"| Analyser | repostats {report.analyser_version} |",
333 ]
334 if report.branch:
335 lines.append(f"| Branch | `{report.branch}` |")
336 if report.head_commit:
337 lines.append(f"| HEAD | `{report.head_commit[:12]}` |")
338 if report.remote:
339 lines.append(f"| Remote | {report.remote} |")
340 lines.append("")
341
342 if report.warnings:
343 lines.append("> **Caveats**")
344 for warning in report.warnings[:8]:
345 lines.append(f"> - {warning}")
346 lines.append("")
347 return lines

350def render_markdown(report: RepoReport) -> str:
351 sections = [
352 _header(report),
353 _section_languages(report),
354 _section_build(report),
355 _section_purpose(report),
356 _section_contributors(report),
357 _section_flow(report),
358 _section_dependencies(report),
359 _section_code_stats(report),
360 _section_testing(report),
361 [
362 "---",
363 "",
364 "*Sections 3 and 5 are inferred from naming conventions, package metadata and "
365 "declared dependencies. They describe the architecture the layout implies, which "
366 "is not always the architecture the code actually implements — check the evidence "
367 "blocks before relying on them.*",
368 "",
369 ],
370 ]
371 return "\n".join(line for section in sections for line in section).rstrip() + "\n"
