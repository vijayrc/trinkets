146def _page(title: str, body: str, *, extra_style: str = "") -> str:
147 return (
148 "<!doctype html>\n<html lang=\"en\">\n<head>\n"
149 "<meta charset=\"utf-8\">\n"
150 "<meta name=\"viewport\" content=\"width=device-width, initial-scale=1\">\n"
151 f"<title>{esc(title)}</title>\n"
152 f"<style>{BASE_STYLE}{extra_style}</style>\n"
153 "</head>\n<body>\n<main>\n" + body + "\n</main>\n</body>\n</html>\n"
154 )

157def _pct(part: float, whole: float) -> float:
158 return (part / whole * 100) if whole else 0.0

161def _humanise_bytes(count: int) -> str:
162 size = float(count)
163 for unit in ("B", "KB", "MB", "GB"):
164 if size < 1024 or unit == "GB":
165 return f"{size:.0f} {unit}" if unit == "B" else f"{size:.1f} {unit}"
166 size /= 1024
167 return f"{size:.1f} GB"

170def _badge(confidence: Confidence) -> str:
171 cls = CONFIDENCE_CLASS.get(confidence, "conf-low")
172 label = CONFIDENCE_LABEL.get(confidence, confidence.value)
173 return f'<span class="badge {cls}">{esc(label)}</span>'

175
176def _chips(values: list[str]) -> str:
177 return "".join(f'<span class="chip">{esc(v)}</span>' for v in values)

180def _code(text: str, href: str | None = None) -> str:
181 inner = f"<code>{esc(text)}</code>"
182 return f'<a class="filelink" href="{esc(href)}">{inner}</a>' if href else inner

185def _bar_row(label: str, value: str, pct: float, color: str) -> str:
186 pct = max(0.0, min(100.0, pct))
187 fill = f'<div class="bar-fill" style="width:{pct:.1f}%;background:{color}"></div>'
188 return (
189 '<div class="bar-row">'
190 f'<div class="bar-label" title="{esc(label)}">{esc(label)}</div>'
191 f'<div class="bar-track">{fill}</div>'
192 f'<div class="bar-value">{esc(value)}</div>'
193 "</div>"
194 )

197def _table(
198 headers: list[str], rows: list[list[str]], *, numeric_cols: set[int] = frozenset()
199) -> str:
200 head = "".join(
201 f'<th class="num">{esc(h)}</th>' if i in numeric_cols else f"<th>{esc(h)}</th>"
202 for i, h in enumerate(headers)
203 )
204 body_rows = []
205 for row in rows:
206 cells = "".join(
207 f'<td class="num">{cell}</td>' if i in numeric_cols else f"<td>{cell}</td>"
208 for i, cell in enumerate(row)
209 )
210 body_rows.append(f"<tr>{cells}</tr>")
211 body = "".join(body_rows)
212 return f"<table><thead><tr>{head}</tr></thead><tbody>{body}</tbody></table>"

218def _section_header(report: RepoReport) -> str:
219 rows = [
220 ["Generated", esc(report.generated_at)],
221 ["Analyser", f"repostats {esc(report.analyser_version)}"],
222 ]
223 if report.branch:
224 rows.append(["Branch", _code(report.branch)])
225 if report.head_commit:
226 rows.append(["HEAD", _code(report.head_commit[:12])])
227 if report.remote:
228 rows.append(["Remote", esc(report.remote)])
229 meta = "".join(f"<tr><td>{k}</td><td>{v}</td></tr>" for k, v in rows)
230
231 html = [
232 f"<h1>Repository report: {esc(report.name)}</h1>",
233 f'<div class="path">{esc(report.path)}</div>',
234 f"<table><tbody>{meta}</tbody></table>",
235 ]
236 if report.warnings:
237 items = "".join(f"<li>{esc(w)}</li>" for w in report.warnings[:8])
238 html.append(f'<div class="callout"><strong>Caveats</strong><ul>{items}</ul></div>')
239 return "\n".join(html)

242def _section_languages(report: RepoReport) -> str:
243 code_languages = [entry for entry in report.languages if entry.is_code]
244 other_languages = [entry for entry in report.languages if not entry.is_code]
245 html = ["<h2>1. Programming languages</h2>"]
246
247 if not code_languages:
248 html.append("<p>No recognised source code found.</p>")
249 return "\n".join(html)
250
251 total_code = sum(entry.code_lines for entry in code_languages) or 1
252 ranked = code_languages[:15]
253
254 stack = "".join(
255 f'<div style="width:{_pct(e.code_lines, total_code):.2f}%;'
256 f'background:{PALETTE[i % len(PALETTE)]}"></div>'
257 for i, e in enumerate(ranked)
258 )
259 legend = "".join(
260 f'<span><span class="swatch" style="background:{PALETTE[i % len(PALETTE)]}">'
261 f"</span>{esc(e.language)}</span>"
262 for i, e in enumerate(ranked)
263 )
264 html.append(f'<div class="stack">{stack}</div><div class="legend">{legend}</div>')
265
266 rows = [
267 [
268 esc(e.language), f"{e.files:,}", f"{e.code_lines:,}",
269 f"{_pct(e.code_lines, total_code):.1f}%", f"{e.comment_lines:,}", f"{e.blank_lines:,}",
270 ]
271 for e in ranked
272 ]
273 html.append(
274 _table(
275 ["Language", "Files", "Code lines", "Share", "Comments", "Blank"],
276 rows, numeric_cols={1, 2, 3, 4, 5},
277 )
278 )
279 primary = code_languages[0]
280 html.append(
281 f"<p><strong>Primary language:</strong> {esc(primary.language)} "
282 f"({_pct(primary.code_lines, total_code):.1f}% of source lines).</p>"
283 )
284 if other_languages:
285 summary = ", ".join(f"{esc(e.language)} ({e.files})" for e in other_languages[:8])
286 html.append(f"<p><strong>Config / markup files:</strong> {summary}.</p>")
287 return "\n".join(html)

350def _section_contributors(report: RepoReport) -> str:
351 info = report.contributors
352 html = ["<h2>4. Contributors and commit timeframe</h2>"]
353 if not info.total_commits:
354 html.append("<p>No commit history available.</p>")
355 return "\n".join(html)
356
357 span = "—"
358 if info.first_commit and info.last_commit:
359 span = f"{info.first_commit} → {info.last_commit}"
360 rows = [
361 ["Total commits (excl. merges)", f"{info.total_commits:,}"],
362 ["Distinct authors", f"{info.total_authors:,}"],
363 ["Active span", esc(span)],
364 ["Days with commits", f"{info.active_days:,}"],
365 ["Bus factor", f"{info.bus_factor:,}"],
366 ]
367 html.append(_table(["Metric", "Value"], rows))
368 if info.bus_factor_note:
369 html.append(f"<p><em>{esc(info.bus_factor_note)}</em></p>")
370
371 html.append("<h3>Top contributors</h3>")
372 crows = [
373 [
374 esc(p.name), f"{p.commits:,}", f"{_pct(p.commits, info.total_commits):.1f}%",
375 f"{p.insertions:,}", f"{p.deletions:,}",
376 esc(f"{p.first_commit or '?'} → {p.last_commit or '?'}"),
377 ]
378 for p in info.top_contributors[:10]
379 ]
380 html.append(
381 _table(
382 ["Author", "Commits", "Share", "Insertions", "Deletions", "Active"],
383 crows, numeric_cols={1, 2, 3, 4},
384 )
385 )
386
387 if info.commits_by_year:
388 html.append("<h3>Commits by year</h3>")
389 peak = max(info.commits_by_year.values()) or 1
390 bars = "".join(
391 _bar_row(year, f"{count:,}", _pct(count, peak), PALETTE[i % len(PALETTE)])
392 for i, (year, count) in enumerate(info.commits_by_year.items())
393 )
394 html.append(f'<div class="bars">{bars}</div>')
395 return "\n".join(html)

398def _section_flow(report: RepoReport) -> str:
399 flow = report.flow
400 html = [f"<h2>5. Flow: entry to persistence {_badge(flow.confidence)}</h2>"]
401 if flow.note:
402 html.append(f"<blockquote>{esc(flow.note)}</blockquote>")
403 if flow.mermaid:
404 html.append(f'<pre class="mono">{esc(flow.mermaid)}</pre>')
405 interesting = [node for node in flow.nodes if node.evidence]
406 if interesting:
407 items = []
408 for node in interesting:
409 label = esc(node.label.replace("<br/>", " "))
410 sub = "".join(
411 "<li>" + esc(item.detail)
412 + (f" — <code>{esc(item.path)}</code>" if item.path else "")
413 + "</li>"
414 for item in node.evidence
415 )
416 items.append(f"<li><strong>{label}</strong><ul>{sub}</ul></li>")
417 summary = "How each node was identified"
418 html.append(f"<details><summary>{summary}</summary><ul>{''.join(items)}</ul></details>")
419 return "\n".join(html)

422def _section_dependencies(report: RepoReport, file_href: FileLinker) -> str:
423 deps = report.dependencies
424 html = ["<h2>6. External dependencies and infrastructure</h2>"]
425
426 if deps.by_category:
427 html.append("<h3>Systems this code talks to</h3>")
428 rows = [[esc(cat.title()), _chips(systems)] for cat, systems in deps.by_category.items()]
429 html.append(_table(["Category", "Systems"], rows))
430 else:
431 html.append("<p>No databases, caches, queues or external services detected.</p>")
432
433 if deps.infrastructure:
434 html.append("<h3>Infrastructure evidence</h3>")
435 rows = [
436 [
437 esc(d.name), esc(d.category), esc(d.detail or "—"),
438 _code(d.source, file_href(d.source)),
439 ]
440 for d in deps.infrastructure[:20]
441 ]
442 html.append(_table(["System", "Category", "Evidence", "Source"], rows))
443
444 html.append(
445 f"<p><strong>Declared dependencies:</strong> {deps.total_declared:,} "
446 "across all manifests.</p>"
447 )
448
449 notable = [d for d in deps.declared if d.category != "other"]
450 if notable:
451 rows = [
452 [
453 _code(d.name), esc(d.version or "—"), esc(d.category), esc(d.detail or "—"),
454 _code(d.source, file_href(d.source)),
455 ]
456 for d in sorted(notable, key=lambda d: (d.category, d.name))[:60]
457 ]
458 table = _table(["Package", "Version", "Category", "Implies", "Manifest"], rows)
459 summary = "Classified declared dependencies"
460 html.append(f"<details><summary>{summary}</summary>{table}</details>")
461 return "\n".join(html)

465 code = report.code
466 html = ["<h2>7. Size and structure</h2>"]
467 comment_ratio = _pct(code.comment_lines, code.code_lines + code.comment_lines)
468 rows = [
469 ["Files analysed", f"{code.analysed_files:,}"],
470 ["Files skipped (binary, vendored, generated)", f"{code.skipped_files:,}"],
471 ["Total lines", f"{code.total_lines:,}"],
472 ["Code lines", f"{code.code_lines:,}"],
473 ["Comment lines", f"{code.comment_lines:,}"],
474 ["Blank lines", f"{code.blank_lines:,}"],
475 ["Comment ratio", f"{comment_ratio:.1f}%"],
476 ["Classes / types", f"{code.classes:,}"],
477 ["Functions / methods", f"{code.functions:,}"],
478 ["HTTP endpoints", f"{code.api_endpoints:,}"],
479 ["Repository size (analysed)", esc(_humanise_bytes(code.bytes))],
480 ]
481 html.append(_table(["Metric", "Value"], rows))
482
483 if code.endpoints:
484 html.append("<h3>API endpoints</h3>")
485 rows = []
486 for e in code.endpoints[:40]:
487 href = file_href(e.path)
488 loc_href = f"{href}#L{e.line}" if href else None
489 rows.append(
490 [
491 f"<code>{esc(e.method)}</code>", _code(e.route),
492 _code(f"{e.path}:{e.line}", loc_href), esc(e.framework),
493 ]
494 )
495 html.append(_table(["Method", "Route", "Location", "Detected as"], rows))
496 if len(code.endpoints) > 40:
497 html.append(f"<p><em>{len(code.endpoints) - 40} more not shown.</em></p>")
498
499 if code.counts_by_language:
500 html.append("<h3>Declarations by language</h3>")
501 rows = [
502 [esc(lang), f"{counts['classes']:,}", f"{counts['functions']:,}"]
503 for lang, counts in sorted(
504 code.counts_by_language.items(), key=lambda item: -item[1]["functions"]
505 )
506 ]
507 html.append(_table(["Language", "Classes / types", "Functions"], rows, numeric_cols={1, 2}))
508
509 if code.largest_files:
510 html.append("<h3>Largest source files</h3>")
511 peak = max(count for _, count in code.largest_files) or 1
512 bars = "".join(
513 _largest_file_row(path, count, peak, PALETTE[i % len(PALETTE)], file_href(path))
514 for i, (path, count) in enumerate(code.largest_files)
515 )
516 html.append(f'<div class="bars">{bars}</div>')
517 return "\n".join(html)

527 f'<div class="bar-track">{fill}</div>'
528 f'<div class="bar-value">{count:,}</div>'
529 "</div>"

540 coverage_tools = ", ".join(testing.coverage_tools_configured) or "none"
541 rows = [
542 ["Test frameworks", esc(", ".join(testing.frameworks) or "none detected")],
543 ["Coverage tooling configured", esc(coverage_tools)],
544 ["Test files", f"{testing.test_files:,}"],
545 ["Source files (non-test)", f"{testing.source_files:,}"],
546 ["Test-to-source file ratio", f"{testing.test_to_source_ratio:.2f}"],
547 ["Test functions / cases", f"{testing.test_functions:,}"],
548 ["Lines of test code", f"{testing.test_lines:,}"],
549 ["Line coverage", esc(coverage)],
550 ]
551 html.append(_table(["Metric", "Value"], rows))
552 if testing.coverage_source:
553 html.append(f"<p>Coverage read from <code>{esc(testing.coverage_source)}</code>.</p>")
554 if testing.note:
555 html.append(f"<blockquote>{esc(testing.note)}</blockquote>")
556 return "\n".join(html)

587def render_source_file(
588 relpath: str,
589 text: str,
590 *,
591 repo_name: str,
592 report_href: str,
593 index_href: str,
594 truncated: bool = False,
595) -> str:
596 lines = text.splitlines() or [""]
597 rows = []
598 for n, line in enumerate(lines, start=1):
599 rows.append(f'<tr id="L{n}"><td class="ln">{n}</td><td class="code">{esc(line)}</td></tr>')
600 note = ""
601 if truncated:
602 note = "<p><em>File truncated — showing the first lines only.</em></p>"
603 body = (
604 f'<div class="crumbs"><a href="{esc(index_href)}">&larr; all repositories</a>'
605 f' &nbsp;/&nbsp; <a href="{esc(report_href)}">&larr; {esc(repo_name)} report</a></div>'
606 f"<h1>{esc(relpath)}</h1>"
607 f"{note}"
608 f'<table class="src"><tbody>{"".join(rows)}</tbody></table>'
609 )
610 return _page(f"{relpath} — {repo_name}", body, extra_style=SOURCE_STYLE)

613def render_index(
614 root_name: str,
615 generated_at: str,
616 rows: list[dict],
617 errors: list[tuple[str, str]],
618) -> str:
619 """``rows`` items: name, display_path, report_href, language, code_lines,
620 files, contributors, commits, bus_factor, warnings (count)."""
621 body_rows = []
622 for r in rows:
623 name_cell = f'<a href="{esc(r["report_href"])}"><strong>{esc(r["name"])}</strong></a>'
624 if r["display_path"] and r["display_path"] != r["name"]:
625 name_cell += f'<div class="path">{esc(r["display_path"])}</div>'
626 warn = ""
627 if r["warnings"]:
628 warn = f' <span class="badge conf-low">{r["warnings"]} warning(s)</span>'
629 body_rows.append(
630 "<tr>"
631 f"<td>{name_cell}{warn}</td>"
632 f'<td>{esc(r["language"] or "—")}</td>'
633 f'<td class="num">{r["code_lines"]:,}</td>'
634 f'<td class="num">{r["files"]:,}</td>'
635 f'<td class="num">{r["contributors"]:,}</td>'
636 f'<td class="num">{r["commits"]:,}</td>'
637 f'<td class="num">{r["bus_factor"]:,}</td>'
638 "</tr>"
639 )
640 table = (
641 "<table><thead><tr><th>Repository</th><th>Primary language</th>"
642 '<th class="num">Code lines</th><th class="num">Files</th>'
643 '<th class="num">Contributors</th><th class="num">Commits</th>'
644 '<th class="num">Bus factor</th></tr></thead>'
645 f"<tbody>{''.join(body_rows)}</tbody></table>"
646 )
647
648 error_block = ""
649 if errors:
650 items = "".join(f"<li><code>{esc(path)}</code> — {esc(msg)}</li>" for path, msg in errors)
651 error_block = (
652 f'<div class="callout"><strong>Could not be analysed</strong><ul>{items}</ul></div>'
653 )
654
655 repo_word = "repository" if len(rows) == 1 else "repositories"
656 body = (
657 f"<h1>Repository scan: {esc(root_name)}</h1>"
658 f'<p class="path">Generated {esc(generated_at)} &middot; {len(rows)} {repo_word}</p>'
659 f"{table}{error_block}"
660 )
661 return _page(f"Repository scan: {root_name} — repostats", body)
