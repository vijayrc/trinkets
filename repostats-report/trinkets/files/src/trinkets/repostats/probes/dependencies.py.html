25_REQUIREMENT_LINE = re.compile(r"^\s*([A-Za-z0-9][A-Za-z0-9._-]*)\s*(\[[^\]]*\])?\s*([<>=!~^].*)?$")
26_PEP508_NAME = re.compile(r"^\s*([A-Za-z0-9][A-Za-z0-9._-]*)")
27_GRADLE_DEP = re.compile(
28 r"""(?:implementation|api|compile|runtimeOnly|testImplementation|testCompile|
29 compileOnly|annotationProcessor|kapt)\s*\(?\s*['"]([^'"]+)['"]""",
30 re.VERBOSE,
31)
32_GO_REQUIRE = re.compile(r"^\s*([a-zA-Z0-9._~/-]+\.[a-z]{2,}/[^\s]+)\s+v[\d]", re.MULTILINE)
33_DOCKER_FROM = re.compile(r"^\s*FROM\s+([^\s]+)", re.MULTILINE | re.IGNORECASE)
34_COMPOSE_IMAGE = re.compile(r"^\s*image:\s*['\"]?([^'\"\s#]+)", re.MULTILINE)
35_PY_IMPORT = re.compile(r"^\s*(?:from|import)\s+([a-zA-Z_][\w.]*)", re.MULTILINE)
36_JS_IMPORT = re.compile(r"""(?:from|require\()\s*['"]([^'"./][^'"]*)['"]""")
37_JAVA_IMPORT = re.compile(r"^\s*import\s+(?:static\s+)?([\w.]+)", re.MULTILINE)

40def _strip_pep508(spec: str) -> str | None:
41 """'requests[socks] >=2,<3 ; python_version<"3.9"' -> 'requests'."""
42 spec = spec.split(";", 1)[0].strip()
43 if not spec or spec.startswith(("#", "-", ".", "/")):
44 return None
45 match = _PEP508_NAME.match(spec)
46 return match.group(1) if match else None

49def _from_pyproject(source: SourceFile) -> list[tuple[str, str | None]]:
50 if not source.text:
51 return []
52 try:
53 data = tomllib.loads(source.text)
54 except (tomllib.TOMLDecodeError, ValueError):
55 return []
56
57 found: list[tuple[str, str | None]] = []
58 project = data.get("project", {})
59 for spec in project.get("dependencies", []) or []:
60 name = _strip_pep508(str(spec))
61 if name:
62 found.append((name, None))
63 for extras in (project.get("optional-dependencies") or {}).values():
64 for spec in extras or []:
65 name = _strip_pep508(str(spec))
66 if name:
67 found.append((name, None))
68
69 poetry = (data.get("tool") or {}).get("poetry") or {}
70 for group in ("dependencies", "dev-dependencies"):
71 for name, constraint in (poetry.get(group) or {}).items():
72 if name.lower() != "python":
73 version = constraint if isinstance(constraint, str) else None
74 found.append((name, version))
75 for group_data in (poetry.get("group") or {}).values():
76 for name, constraint in (group_data.get("dependencies") or {}).items():
77 if name.lower() != "python":
78 found.append((name, constraint if isinstance(constraint, str) else None))
79 return found

82def _from_requirements(source: SourceFile) -> list[tuple[str, str | None]]:
83 found: list[tuple[str, str | None]] = []
84 for line in source.lines:
85 stripped = line.strip()
86 if not stripped or stripped.startswith(("#", "-r", "--", "-e", "git+", "http")):
87 continue
88 name = _strip_pep508(stripped)
89 if not name:
90 continue
91 match = _REQUIREMENT_LINE.match(stripped)
92 version = match.group(3).strip() if match and match.group(3) else None
93 found.append((name, version))
94 return found

97def _from_package_json(source: SourceFile) -> list[tuple[str, str | None]]:
98 if not source.text:
99 return []
100 try:
101 data = json.loads(source.text)
102 except (json.JSONDecodeError, ValueError):
103 return []
104 if not isinstance(data, dict):
105 return []
106 found: list[tuple[str, str | None]] = []
107 for section in ("dependencies", "devDependencies", "peerDependencies", "optionalDependencies"):
108 for name, version in (data.get(section) or {}).items():
109 found.append((name, str(version) if version else None))
110 return found

113def _from_pom(source: SourceFile) -> list[tuple[str, str | None]]:
114 if not source.text:
115 return []
116 found: list[tuple[str, str | None]] = []
117 for block in re.findall(r"<dependency>(.*?)</dependency>", source.text, re.DOTALL):
118 artifact = re.search(r"<artifactId>\s*([^<]+?)\s*</artifactId>", block)
119 version = re.search(r"<version>\s*([^<]+?)\s*</version>", block)
120 if artifact:
121 found.append((artifact.group(1), version.group(1) if version else None))
122 return found

125def _from_gradle(source: SourceFile) -> list[tuple[str, str | None]]:
126 if not source.text:
127 return []
128 found: list[tuple[str, str | None]] = []
129 for coordinate in _GRADLE_DEP.findall(source.text):
130 pieces = coordinate.split(":")
131 if len(pieces) >= 2:
132 found.append((pieces[1], pieces[2] if len(pieces) > 2 else None))
133 else:
134 found.append((coordinate, None))
135 return found

192def _compose_services(source: SourceFile) -> list[tuple[str, str]]:
193 """Return (service name, image) pairs from a compose file."""
194 if not source.text:
195 return []
196 if yaml is not None:
197 try:
198 data = yaml.safe_load(source.text)
199 if isinstance(data, dict) and isinstance(data.get("services"), dict):
200 pairs = []
201 for name, spec in data["services"].items():
202 if isinstance(spec, dict) and spec.get("image"):
203 pairs.append((str(name), str(spec["image"])))
204 return pairs
205 except Exception: # noqa: BLE001 - malformed YAML shouldn't kill the run
206 pass
207 return [("", image) for image in _COMPOSE_IMAGE.findall(source.text)]

210def analyse(files: list[SourceFile]) -> DependencyInfo:
211 info = DependencyInfo()
212 seen_declared: set[tuple[str, str]] = set()
213 seen_infra: set[str] = set()
214
215 # --- 1. declared dependencies ---------------------------------------
216 for source in files:
217 name = source.name.lower()
218 parser = PARSERS.get(name)
219 if parser is None and (name.startswith("requirements") and name.endswith(".txt")):
220 parser = _from_requirements
221 if parser is None:
222 continue
223
224 for dep_name, version in parser(source):
225 key = (knowledge.normalise(dep_name), source.posix_relpath)
226 if key in seen_declared:
227 continue
228 seen_declared.add(key)
229
230 classified = knowledge.classify_package(dep_name)
231 category, detail = classified if classified else ("other", None)
232 info.declared.append(
233 Dependency(
234 name=dep_name,
235 category=category,
236 source=source.posix_relpath,
237 version=version,
238 detail=detail,
239 confidence=Confidence.HIGH,
240 )
241 )
242
243 info.total_declared = len(info.declared)
244
245 # --- 2. container images --------------------------------------------
246 for source in files:
247 name = source.name.lower()
248 is_compose = name.startswith(("docker-compose", "compose")) and name.endswith(
249 (".yml", ".yaml")
250 )
251 is_dockerfile = name == "dockerfile" or name.startswith("dockerfile.")
252
253 images: list[tuple[str, str]] = []
254 if is_compose:
255 images = _compose_services(source)
256 elif is_dockerfile and source.text:
257 images = [("", image) for image in _DOCKER_FROM.findall(source.text)]
258
259 for service_name, image in images:
260 classified = knowledge.classify_image(image)
261 if not classified:
262 continue
263 category, system = classified
264 if system in seen_infra:
265 continue
266 seen_infra.add(system)
267 label = f"{service_name} ({image})" if service_name else image
268 info.infrastructure.append(
269 Dependency(
270 name=system,
271 category=category,
272 source=source.posix_relpath,
273 detail=f"container image: {label}",
274 confidence=Confidence.HIGH,
275 )
276 )
277
278 # --- 3. imports in source -------------------------------------------
279 import_systems: dict[str, tuple[str, str]] = {}
280 for source in files:
281 if not source.text or not source.is_code:
282 continue
283 modules: list[str] = []
284 if source.language in {"Python", "Cython"}:
285 modules = _PY_IMPORT.findall(source.text)
286 elif source.language in {"JavaScript", "TypeScript"}:
287 modules = _JS_IMPORT.findall(source.text)
288 elif source.language in {"Java", "Kotlin", "Scala"}:
289 modules = _JAVA_IMPORT.findall(source.text)
290 if not modules:
291 continue
292
293 for module in modules:
294 root = module.split(".", 1)[0].lower()
295 for fragment, category, system in knowledge.IMPORT_SYSTEMS:
296 if fragment in root:
297 import_systems.setdefault(system, (category, source.posix_relpath))
298 break
299
300 declared_systems = {dep.detail for dep in info.declared if dep.detail}
301 infra_systems = {dep.name for dep in info.infrastructure}
302 for system, (category, path) in sorted(import_systems.items()):
303 if system in declared_systems or system in infra_systems:
304 continue
305 info.infrastructure.append(
306 Dependency(
307 name=system,
308 category=category,
309 source=path,
310 detail="inferred from import statements (not found in any manifest)",
311 confidence=Confidence.MEDIUM,
312 )
313 )
314
315 # --- roll up ---------------------------------------------------------
316 by_category: dict[str, set[str]] = defaultdict(set)
317 for dep in info.declared:
318 if dep.category != "other" and dep.detail:
319 by_category[dep.category].add(dep.detail)
320 for dep in info.infrastructure:
321 by_category[dep.category].add(dep.name)
322
323 info.by_category = {
324 category: sorted(by_category[category])
325 for category in knowledge.CATEGORY_ORDER
326 if by_category.get(category)
327 }
328 return info
