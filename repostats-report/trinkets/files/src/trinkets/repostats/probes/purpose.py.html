1"""Probe 3: inferred purpose of the repository.
2
3This is the most speculative section in the report. It reads what the project
4says about itself (README, package metadata) and corroborates that against what
5the code actually contains, then labels the result with a confidence level.
6It does not attempt to summarise business logic it cannot see.
7"""

19BADGE_LINE = re.compile(r"^\s*(?:\[!\[|!\[|<img|<p\s|<div\s|<h\d|<a\s|=+\s*$|-+\s*$)")
20MD_DECORATION = re.compile(r"[`*_>#]|\[([^\]]*)\]\([^)]*\)")

22STOPWORDS = frozenset({
23 "src", "lib", "app", "apps", "main", "core", "common", "utils", "util", "test", "tests",
24 "docs", "doc", "internal", "pkg", "cmd", "api", "web", "config", "scripts", "build",
25 "index", "types", "helpers", "shared", "base", "public", "static", "assets", "dist",
26 "the", "and", "for", "with", "from", "this", "that", "new", "old", "temp", "tmp",
27})

29CLI_SIGNALS = (
30 "argparse", "click", "typer", "docopt", "commander", "yargs", "oclif",
31 "cobra", "clap", "picocli", "thor",

34KIND_ORDER = (
35 "HTTP service / web API",

37 "CLI tool",
38 "library / SDK",

40 "machine learning",

48def _clean_markdown(line: str) -> str:
49 cleaned = MD_DECORATION.sub(lambda m: m.group(1) or "", line).strip()
50 return re.sub(r"\s+", " ", cleaned)

53def _readme_excerpt(readme: SourceFile | None) -> tuple[str | None, str | None]:
54 """Return (title, first substantive paragraph)."""
55 if readme is None or not readme.text:
56 return None, None
57
58 title: str | None = None
59 paragraph: list[str] = []
60
61 for raw in readme.lines[:120]:
62 line = raw.strip()
63 if not line:
64 if paragraph:
65 break
66 continue
67 if line.startswith("#") and title is None:
68 title = _clean_markdown(line.lstrip("#").strip()) or None
69 continue
70 if BADGE_LINE.match(line):
71 continue
72 if line.startswith(("---", "===", "```", "|", ":--")):
73 if paragraph:
74 break
75 continue
76 cleaned = _clean_markdown(line)
77 if cleaned:
78 paragraph.append(cleaned)
79 if sum(len(part) for part in paragraph) > 400:
80 break
81
82 text = " ".join(paragraph).strip()
83 if len(text) > 500:
84 text = text[:497].rsplit(" ", 1)[0] + "..."
85 return title, text or None

88def _declared_metadata(files: list[SourceFile]) -> tuple[str | None, str | None]:
89 """Return (project name, description) from package metadata."""
90 by_name = {source.posix_relpath: source for source in files}
91
92 pyproject = by_name.get("pyproject.toml")
93 if pyproject and pyproject.text:
94 try:
95 data = tomllib.loads(pyproject.text)
96 project = data.get("project") or {}
97 poetry = (data.get("tool") or {}).get("poetry") or {}
98 name = project.get("name") or poetry.get("name")
99 description = project.get("description") or poetry.get("description")
100 if name or description:
101 return name, description
102 except (tomllib.TOMLDecodeError, ValueError):
103 pass
104
105 package_json = by_name.get("package.json")
106 if package_json and package_json.text:
107 try:
108 data = json.loads(package_json.text)
109 if isinstance(data, dict):
110 return data.get("name"), data.get("description")
111 except (json.JSONDecodeError, ValueError):
112 pass
113
114 pom = by_name.get("pom.xml")
115 if pom and pom.text:
116 name = re.search(r"<artifactId>\s*([^<]+?)\s*</artifactId>", pom.text)
117 description = re.search(r"<description>\s*([^<]+?)\s*</description>", pom.text, re.DOTALL)
118 if name or description:
119 return (
120 name.group(1) if name else None,
121 re.sub(r"\s+", " ", description.group(1)).strip() if description else None,
122 )
123
124 cargo = by_name.get("Cargo.toml")
125 if cargo and cargo.text:
126 try:
127 data = tomllib.loads(cargo.text)
128 package = data.get("package") or {}
129 return package.get("name"), package.get("description")
130 except (tomllib.TOMLDecodeError, ValueError):
131 pass
132
133 return None, None

136def _domain_terms(files: list[SourceFile], limit: int = 12) -> list[str]:
137 """Recurring identifiers in paths — a rough proxy for the problem domain."""
138 counter: Counter[str] = Counter()
139 for source in files:
140 if not source.is_code:
141 continue
142 parts = source.posix_relpath.replace("\\", "/").split("/")
143 stem = parts[-1].rsplit(".", 1)[0]
144 for token in [*parts[:-1], stem]:
145 for word in re.split(r"[-_.\s]+|(?<=[a-z])(?=[A-Z])", token):
146 word = word.strip().lower()
147 if len(word) < 4 or word in STOPWORDS or word.isdigit():
148 continue
149 counter[word] += 1
150 return [word for word, count in counter.most_common(limit) if count >= 2]

153def analyse(
154 files: list[SourceFile],
155 build: BuildInfo,
156 endpoints: list[ApiEndpoint],
157 frameworks: list[str],
158 languages_ranked: list[str],
159) -> PurposeInfo:
160 info = PurposeInfo()
161 evidence: list[Evidence] = []
162
163 readme = next(
164 (source for source in files
165 if source.name.lower().startswith("readme") and source.posix_relpath.count("/") == 0),
166 None,
167 )
168 title, excerpt = _readme_excerpt(readme)
169 info.readme_excerpt = excerpt
170 if readme is not None:
171 evidence.append(Evidence("Read project README", readme.posix_relpath))
172
173 name, description = _declared_metadata(files)
174 info.project_name = name or title
175 info.declared_description = description
176 if description:
177 evidence.append(Evidence("Package metadata declares a description"))
178
179 info.frameworks = frameworks
180
181 # --- what kind of thing is this? ------------------------------------
182 kinds: set[str] = set()
183 all_text_names = {source.posix_relpath.lower() for source in files}
184
185 if endpoints:
186 kinds.add("HTTP service / web API")
187 evidence.append(Evidence(f"Found {len(endpoints)} HTTP route declarations"))
188
189 web_frontend = {"React", "Vue", "Angular", "Svelte", "Next.js", "Nuxt"}
190 if any(framework in web_frontend for framework in frameworks):
191 kinds.add("web frontend")
192 if any(name.endswith((".tsx", ".jsx", ".vue", ".svelte")) for name in all_text_names):
193 kinds.add("web frontend")
194
195 if build.tools and any("script" in tool.lower() for tool in build.tools):
196 pass
197 has_console_script = False
198 for source in files:
199 if source.name.lower() == "pyproject.toml" and source.text:
200 if "[project.scripts]" in source.text or "console_scripts" in source.text:
201 has_console_script = True
202 if source.name.lower() == "package.json" and source.text and '"bin"' in source.text:
203 has_console_script = True
204 if has_console_script:
205 kinds.add("CLI tool")
206 evidence.append(Evidence("Package metadata declares an executable entry point"))
207
208 code_text_sample = " ".join(
209 source.text[:4000] for source in files
210 if source.text and source.is_code and source.posix_relpath.count("/") <= 3
211 ).lower()
212 for signal in CLI_SIGNALS:
213 if signal in code_text_sample:
214 kinds.add("CLI tool")
215 evidence.append(Evidence(f"CLI framework referenced in source: {signal}"))
216 break
217
218 if any(name.endswith(".tf") for name in all_text_names):
219 kinds.add("infrastructure as code")
220 if any(name.endswith(".ipynb") for name in all_text_names):
221 kinds.add("data pipeline / analytics")
222 if any(framework in {"pandas", "NumPy", "Apache Spark", "Dask", "Polars"}
223 for framework in frameworks):
224 kinds.add("data pipeline / analytics")
225 if any(framework in {"PyTorch", "TensorFlow", "scikit-learn", "Hugging Face Transformers",
226 "LangChain", "OpenAI API", "Anthropic API"} for framework in frameworks):
227 kinds.add("machine learning")
228 if any(name.endswith((".swift", ".kt")) and "android" in name for name in all_text_names):
229 kinds.add("mobile app")
230 if len(build.workspaces) >= 3:
231 kinds.add("monorepo of services")
232
233 is_publishable = bool(
234 {"pyproject.toml", "setup.py", "package.json", "Cargo.toml"} & {
235 source.posix_relpath for source in files
236 }
237 )
238 if is_publishable and not kinds:
239 kinds.add("library / SDK")
240
241 info.detected_kinds = [kind for kind in KIND_ORDER if kind in kinds]
242 info.domain_terms = _domain_terms(files)
243
244 # --- assemble a summary ---------------------------------------------
245 subject = info.project_name or "This repository"
246 primary_language = languages_ranked[0] if languages_ranked else "an unidentified language"
247
248 if description:
249 summary = f"{subject} — {description.rstrip('.')}."
250 info.confidence = Confidence.HIGH
251 elif excerpt:
252 summary = f"{subject} — {excerpt.rstrip('.')}."
253 info.confidence = Confidence.MEDIUM
254 else:
255 summary = f"{subject} is a {primary_language} project."
256 info.confidence = Confidence.LOW
257
258 detail_parts: list[str] = []
259 if info.detected_kinds:
260 detail_parts.append("It looks like a " + " and ".join(info.detected_kinds).lower())
261 if frameworks:
262 detail_parts.append(f"built with {', '.join(frameworks[:5])}")
263 if primary_language and not description:
264 detail_parts.append(f"written mainly in {primary_language}")
265 if detail_parts:
266 summary += " " + ", ".join(detail_parts) + "."
267
268 info.summary = summary
269 info.evidence = evidence
270 return info
