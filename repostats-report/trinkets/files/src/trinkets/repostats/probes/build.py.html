90def _read_toml(source: SourceFile) -> dict | None:
91 if not source.text:
92 return None
93 try:
94 return tomllib.loads(source.text)
95 except (tomllib.TOMLDecodeError, ValueError):
96 return None
97
98
99def _read_json(source: SourceFile) -> dict | None:
100 if not source.text:
101 return None
102 try:
103 parsed = json.loads(source.text)
104 except (json.JSONDecodeError, ValueError):
105 return None
106 return parsed if isinstance(parsed, dict) else None

109def analyse(files: list[SourceFile]) -> BuildInfo:
110 info = BuildInfo()
111 tools: set[str] = set()
112 managers: set[str] = set()
113 ci: set[str] = set()
114 workspaces: list[str] = []
115
116 by_relpath = {source.posix_relpath: source for source in files}
117
118 for source in files:
119 rel = source.posix_relpath
120 name = source.name.lower()
121 depth = rel.count("/")
122
123 if name in LOCKFILE_MANAGERS:
124 managers.add(LOCKFILE_MANAGERS[name])
125
126 marker = MARKERS.get(name)
127 if marker:
128 # Only treat deeply-nested manifests as sub-projects, not as the
129 # repo's own build system.
130 if depth <= 2:
131 tool, manager = marker
132 tools.add(tool)
133 if manager:
134 managers.add(manager)
135 info.manifests.append(rel)
136 info.evidence.append(Evidence(f"Found {source.name}", rel))
137 elif name in {"package.json", "pom.xml", "pyproject.toml", "go.mod", "cargo.toml"}:
138 workspaces.append(rel)
139
140 if name in {"dockerfile", "containerfile"} or name.startswith("dockerfile."):
141 info.containerised = True
142
143 lowered_rel = rel.lower()
144 for fragment, system in CI_MARKERS:
145 if lowered_rel.startswith(fragment) or lowered_rel == fragment or fragment in lowered_rel:
146 ci.add(system)
147
148 # --- refine using manifest contents ---------------------------------
149 pyproject = by_relpath.get("pyproject.toml")
150 if pyproject:
151 data = _read_toml(pyproject) or {}
152 backend = (data.get("build-system") or {}).get("build-backend", "")
153 for fragment, label in PEP517_BACKENDS.items():
154 if fragment in str(backend):
155 info.backend = label
156 tools.discard("Python packaging")
157 tools.add(label)
158 break
159 if "poetry" in str(data.get("tool", {}).keys()):
160 managers.add("poetry")
161 if "uv" in data.get("tool", {}):
162 managers.add("uv")
163
164 package_json = by_relpath.get("package.json")
165 if package_json:
166 data = _read_json(package_json) or {}
167 declared_workspaces = data.get("workspaces")
168 if isinstance(declared_workspaces, dict):
169 declared_workspaces = declared_workspaces.get("packages", [])
170 if isinstance(declared_workspaces, list) and declared_workspaces:
171 workspaces.extend(str(item) for item in declared_workspaces)
172 manager_field = data.get("packageManager")
173 if isinstance(manager_field, str) and manager_field:
174 managers.add(manager_field.split("@", 1)[0])
175 for script_tool in ("vite", "webpack", "rollup", "esbuild", "parcel", "tsc", "turbo"):
176 dev_deps = {**data.get("devDependencies", {}), **data.get("dependencies", {})}
177 if script_tool in dev_deps:
178 tools.add(script_tool)
179
180 if "pnpm-workspace.yaml" in by_relpath:
181 managers.add("pnpm")
182 workspaces.append("pnpm-workspace.yaml")
183
184 cargo = by_relpath.get("Cargo.toml") or by_relpath.get("cargo.toml")
185 if cargo:
186 data = _read_toml(cargo) or {}
187 members = (data.get("workspace") or {}).get("members")
188 if isinstance(members, list):
189 workspaces.extend(str(member) for member in members)
190
191 root_pom = by_relpath.get("pom.xml")
192 if root_pom and root_pom.text:
193 modules = re.findall(r"<module>\s*([^<]+?)\s*</module>", root_pom.text)
194 workspaces.extend(modules)
195
196 # --- project model ---------------------------------------------------
197 nested_manifests = [
198 source.posix_relpath
199 for source in files
200 if source.name.lower() in {"package.json", "pom.xml", "pyproject.toml", "go.mod",
201 "cargo.toml", "build.gradle", "build.gradle.kts"}
202 and source.posix_relpath.count("/") >= 1
203 ]
204
205 unique_workspaces = sorted({w for w in workspaces if w})
206 info.workspaces = unique_workspaces
207
208 if unique_workspaces:
209 info.model = f"monorepo / multi-module ({len(unique_workspaces)} declared members)"
210 elif len(nested_manifests) >= 3:
211 info.model = f"multi-project layout ({len(nested_manifests)} nested manifests)"
212 elif tools:
213 info.model = "single project"
214 else:
215 info.model = "no build system detected"
216
217 if info.containerised:
218 info.model += ", containerised"
219
220 info.tools = sorted(tools)
221 info.package_managers = sorted(managers)
222 info.ci_systems = sorted(ci)
223 return info
