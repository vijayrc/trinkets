61def _is_test_file(source: SourceFile) -> bool:
62 parts = [part.lower() for part in Path(source.posix_relpath).parts[:-1]]
63 if any(part in TEST_PATH_MARKERS for part in parts):
64 return True
65 return bool(TEST_FILENAME.match(source.name))

68def _parse_cobertura(text: str) -> float | None:
69 """coverage.xml from coverage.py / Cobertura: line-rate on the root element."""
70 try:
71 root = ET.fromstring(text)
72 except ET.ParseError:
73 return None
74 rate = root.get("line-rate")
75 if rate:
76 try:
77 return round(float(rate) * 100, 2)
78 except ValueError:
79 return None
80 # JaCoCo: <counter type="LINE" missed="x" covered="y"/> at report level
81 for counter in root.findall("counter"):
82 if counter.get("type") == "LINE":
83 missed = int(counter.get("missed", 0))
84 covered = int(counter.get("covered", 0))
85 total = missed + covered
86 if total:
87 return round(covered / total * 100, 2)
88 return None

91def _parse_coverage_json(text: str) -> float | None:
92 try:
93 data = json.loads(text)
94 except (json.JSONDecodeError, ValueError):
95 return None
96 if not isinstance(data, dict):
97 return None
98 totals = data.get("totals")
99 if isinstance(totals, dict):
100 percent = totals.get("percent_covered")
101 if isinstance(percent, int | float):
102 return round(float(percent), 2)
103 # Istanbul summary format
104 total = data.get("total")
105 if isinstance(total, dict) and isinstance(total.get("lines"), dict):
106 percent = total["lines"].get("pct")
107 if isinstance(percent, int | float):
108 return round(float(percent), 2)
109 return None

112def _parse_lcov(text: str) -> float | None:
113 found = hit = 0
114 for line in text.splitlines():
115 if line.startswith("LF:"):
116 found += int(line[3:] or 0)
117 elif line.startswith("LH:"):
118 hit += int(line[3:] or 0)
119 return round(hit / found * 100, 2) if found else None

134def _find_coverage(files: list[SourceFile], repo_path: Path) -> tuple[float | None, str | None]:
135 # Reports are often gitignored, so check the filesystem too.
136 candidates: list[tuple[Path, str, str]] = []
137 for source in files:
138 for filename, kind in COVERAGE_PARSERS:
139 if source.name.lower() == filename:
140 candidates.append((source.abspath, kind, source.posix_relpath))
141
142 for filename, kind in COVERAGE_PARSERS:
143 for location in (repo_path / filename, repo_path / "coverage" / filename,
144 repo_path / "htmlcov" / filename, repo_path / "build" / "reports" / filename):
145 if location.is_file():
146 candidates.append((location, kind, str(location.relative_to(repo_path))))
147
148 for path, kind, relpath in candidates:
149 try:
150 text = path.read_text(encoding="utf-8", errors="replace")
151 except OSError:
152 continue
153 if kind == "cobertura":
154 percent = _parse_cobertura(text)
155 elif kind == "json":
156 percent = _parse_coverage_json(text)
157 else:
158 percent = _parse_lcov(text)
159 if percent is not None:
160 return percent, relpath
161 return None, None

164def _run_coverage(repo_path: Path, timeout: int = 600) -> tuple[float | None, str | None, str]:
165 """Opt-in: actually execute the repo's pytest suite under coverage."""
166 command = ["python", "-m", "pytest", "--cov", "--cov-report=json:.repostats-coverage.json", "-q"]
167 try:
168 subprocess.run(
169 command, cwd=repo_path, capture_output=True, text=True, timeout=timeout, check=False
170 )
171 except (OSError, subprocess.TimeoutExpired) as exc:
172 return None, None, f"Coverage run failed: {exc}"
173
174 report = repo_path / ".repostats-coverage.json"
175 if not report.is_file():
176 return None, None, "Coverage run produced no JSON report (is pytest-cov installed?)."
177 try:
178 percent = _parse_coverage_json(report.read_text(encoding="utf-8", errors="replace"))
179 except OSError as exc:
180 return None, None, f"Could not read coverage report: {exc}"
181 finally:
182 report.unlink(missing_ok=True)
183 return percent, "pytest --cov (executed by repostats)", ""

186def analyse(
187 files: list[SourceFile],
188 repo_path: Path,
189 declared_test_tools: list[str] | None = None,
190 run_coverage: bool = False,
191) -> TestingInfo:
192 info = TestingInfo()
193 frameworks: set[str] = set(declared_test_tools or [])
194 coverage_tools: set[str] = set()
195
196 test_files: list[SourceFile] = []
197 source_files: list[SourceFile] = []
198
199 for source in files:
200 if not source.is_code:
201 if source.name.lower() in COVERAGE_TOOL_FILES:
202 coverage_tools.add(COVERAGE_TOOL_FILES[source.name.lower()])
203 continue
204 if _is_test_file(source):
205 test_files.append(source)
206 else:
207 source_files.append(source)
208
209 # Framework detection from config filenames and in-file usage.
210 filenames = {source.name.lower() for source in files}
211 for framework, (config_files, fragments) in FRAMEWORK_SIGNALS.items():
212 if any(config.lower() in filenames for config in config_files):
213 frameworks.add(framework)
214 continue
215 if not fragments:
216 continue
217 for source in test_files:
218 if source.text and any(fragment in source.text for fragment in fragments):
219 frameworks.add(framework)
220 break
221
222 for source in files:
223 name = source.name.lower()
224 if name in COVERAGE_TOOL_FILES:
225 coverage_tools.add(COVERAGE_TOOL_FILES[name])
226 if name == "pyproject.toml" and source.text and "[tool.coverage" in source.text:
227 coverage_tools.add("coverage.py")
228 if name == "package.json" and source.text:
229 if '"nyc"' in source.text:
230 coverage_tools.add("nyc")
231 if "c8" in source.text:
232 coverage_tools.add("c8")
233
234 info.frameworks = sorted(frameworks)
235 info.coverage_tools_configured = sorted(coverage_tools)
236 info.test_files = len(test_files)
237 info.source_files = len(source_files)
238 info.test_lines = sum(source.code_lines for source in test_files)
239 info.test_to_source_ratio = (
240 round(len(test_files) / len(source_files), 3) if source_files else 0.0
241 )
242
243 test_function = re.compile(
244 r"^\s*(?:async\s+)?def\s+test_\w+|^\s*(?:it|test|describe)\s*\(|@Test\b|^\s*func\s+Test\w+"
245 r"|#\[test\]",
246 re.MULTILINE,
247 )
248 info.test_functions = sum(
249 len(test_function.findall(source.text)) for source in test_files if source.text
250 )
251
252 for source in test_files[:5]:
253 info.evidence.append(Evidence("Test file", source.posix_relpath))
254
255 if run_coverage:
256 percent, label, error = _run_coverage(repo_path)
257 if percent is not None:
258 info.coverage_percent = percent
259 info.coverage_source = label
260 elif error:
261 info.note = error
262
263 if info.coverage_percent is None:
264 percent, source_label = _find_coverage(files, repo_path)
265 info.coverage_percent = percent
266 info.coverage_source = source_label
267
268 if info.coverage_percent is None and not info.note:
269 if coverage_tools:
270 info.note = (
271 f"Coverage tooling is configured ({', '.join(sorted(coverage_tools))}) but no "
272 "coverage report was found in the tree. Run the suite, or pass --run-coverage "
273 "to let repostats execute pytest itself."
274 )
275 else:
276 info.note = (
277 "No coverage report or coverage tooling found. Coverage is not measured here "
278 "because doing so means executing the repository's test suite."
279 )
280
281 return info
