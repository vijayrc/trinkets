1"""Probe 5: request/data flow as a Mermaid diagram.
2
3There is no reliable way to recover true call graphs across every language, so
4this builds a *layer* diagram instead: it classifies files into architectural
5roles using path and filename conventions, corroborates the endpoints of the
6flow (entry points, datastores) with hard evidence from other probes, and wires
7the layers together in their conventional order.
8
9The result is an informed sketch of the intended architecture, not a verified
10call graph — the report says so, and every node carries its evidence.
11"""
12

80def _sanitise(text: str) -> str:
81 """Make a string safe to embed in a Mermaid node label."""
82 cleaned = text.replace('"', "'").replace("[", "(").replace("]", ")")
83 cleaned = cleaned.replace("{", "(").replace("}", ")").replace("|", "/")
84 return cleaned.strip()

87def _classify_layers(files: list[SourceFile]) -> dict[str, list[str]]:
88 """Map layer key -> example file paths."""
89 found: dict[str, list[str]] = {}
90 for source in files:
91 if not source.is_code:
92 continue
93 rel = source.posix_relpath.lower()
94 parts = rel.split("/")
95 stem = parts[-1].rsplit(".", 1)[0]
96 # Split identifiers so userService.ts and user_service.py both match.
97 tokens = set()
98 for part in parts[:-1]:
99 tokens.update(re.split(r"[-_.]+", part))
100 tokens.update(re.split(r"[-_.]+|(?<=[a-z])(?=[A-Z])", stem.lower()))
101
102 for key, _label, fragments in LAYER_RULES:
103 if tokens & set(fragments):
104 found.setdefault(key, []).append(source.posix_relpath)
105 break
106 return found

109def _find_entrypoints(files: list[SourceFile]) -> list[str]:
110 entries: list[tuple[int, str]] = []
111 for source in files:
112 if not source.is_code:
113 continue
114 rel = source.posix_relpath
115 lowered = rel.lower()
116 name = source.name.lower()
117 depth = rel.count("/")
118
119 score = 0
120 if name in ENTRYPOINT_FILENAMES:
121 score += 10
122 if any(hint in lowered for hint in ENTRYPOINT_PATH_HINTS):
123 score += 6
124 if source.text and MAIN_GUARD.search(source.text):
125 score += 8
126 if score:
127 score -= depth # prefer shallower files
128 entries.append((score, rel))
129
130 entries.sort(key=lambda item: (-item[0], item[1]))
131 return [rel for _score, rel in entries[:4]]

134def _node_id(prefix: str, index: int = 0) -> str:
135 return f"{prefix}{index}" if index else prefix

138def analyse(
139 files: list[SourceFile],
140 endpoints: list[ApiEndpoint],
141 dependencies: DependencyInfo,
142 frameworks: list[str],
143) -> FlowInfo:
144 info = FlowInfo()
145 layers = _classify_layers(files)
146 entrypoints = _find_entrypoints(files)
147
148 nodes: list[FlowNode] = []
149 edges: list[FlowEdge] = []
150 signals = 0
151
152 # --- 1. inbound edge -------------------------------------------------
153 if endpoints:
154 caller_label = "HTTP client"
155 caller_detail = f"{len(endpoints)} route(s) declared"
156 elif "CLI" in " ".join(frameworks) or any("cli" in path.lower() for path in entrypoints):
157 caller_label = "User (CLI)"
158 caller_detail = "command line invocation"
159 else:
160 caller_label = "Caller"
161 caller_detail = "entry into the program"
162 nodes.append(
163 FlowNode("caller", caller_label, "external", "stadium",
164 [Evidence(caller_detail)])
165 )
166
167 # --- 2. entry points -------------------------------------------------
168 if entrypoints:
169 signals += 1
170 label = "Entry point<br/>" + "<br/>".join(_sanitise(path) for path in entrypoints[:2])
171 nodes.append(
172 FlowNode("entry", label, "entry", "rect",
173 [Evidence("Entry point", path) for path in entrypoints[:MAX_EVIDENCE_PER_NODE]])
174 )
175 edges.append(FlowEdge("caller", "entry"))
176 upstream = "entry"
177 else:
178 upstream = "caller"
179
180 # --- 3. request-path layers -----------------------------------------
181 for key, label, _fragments in LAYER_RULES:
182 if key in {"worker", "client", "model"}:
183 continue # wired separately below
184 paths = layers.get(key)
185 if not paths:
186 continue
187 signals += 1
188 node_label = f"{label}<br/>({len(paths)} file(s))"
189 nodes.append(
190 FlowNode(key, node_label, key, "rect",
191 [Evidence("Matched by path convention", path)
192 for path in sorted(paths)[:MAX_EVIDENCE_PER_NODE]])
193 )
194 edges.append(FlowEdge(upstream, key))
195 upstream = key
196
197 # If routes exist but no controller directory matched, add a synthetic
198 # handler node so the diagram doesn't skip straight from entry to storage.
199 if endpoints and "controller" not in layers:
200 signals += 1
201 by_file: dict[str, int] = {}
202 for endpoint in endpoints:
203 by_file[endpoint.path] = by_file.get(endpoint.path, 0) + 1
204 top = sorted(by_file.items(), key=lambda item: -item[1])[:MAX_EVIDENCE_PER_NODE]
205 nodes.append(
206 FlowNode("handlers", f"Route handlers<br/>({len(endpoints)} endpoints)", "controller",
207 "rect",
208 [Evidence(f"{count} route(s)", path) for path, count in top])
209 )
210 edges.append(FlowEdge(upstream, "handlers"))
211 upstream = "handlers"
212
213 # Models sit beside the data-access layer rather than in the chain.
214 if "model" in layers:
215 paths = layers["model"]
216 nodes.append(
217 FlowNode("model", f"Models / entities<br/>({len(paths)} file(s))", "model", "rect",
218 [Evidence("Matched by path convention", path)
219 for path in sorted(paths)[:MAX_EVIDENCE_PER_NODE]])
220 )
221 anchor = "repository" if "repository" in layers else upstream
222 edges.append(FlowEdge(anchor, "model", "maps to"))
223
224 # --- 4. persistence and infrastructure sinks -------------------------
225 sink_index = 0
226 persisted = False
227 for category, prefix, shape in SINK_CATEGORIES:
228 systems = dependencies.by_category.get(category)
229 if not systems:
230 continue
231 for system in systems[:4]:
232 sink_index += 1
233 node_id = _node_id(prefix, sink_index)
234 evidence = [
235 Evidence(f"Declared dependency implying {system}", dep.source)
236 for dep in (*dependencies.declared, *dependencies.infrastructure)
237 if dep.detail == system or dep.name == system
238 ][:MAX_EVIDENCE_PER_NODE]
239 nodes.append(FlowNode(node_id, _sanitise(system), category, shape, evidence))
240 edges.append(FlowEdge(upstream, node_id, category))
241 signals += 1
242 if category == "database":
243 persisted = True
244
245 # --- 5. side channels ------------------------------------------------
246 if "worker" in layers:
247 paths = layers["worker"]
248 nodes.append(
249 FlowNode("worker", f"Workers / consumers<br/>({len(paths)} file(s))", "worker", "rect",
250 [Evidence("Matched by path convention", path)
251 for path in sorted(paths)[:MAX_EVIDENCE_PER_NODE]])
252 )
253 queue_node = next((node.node_id for node in nodes if node.layer in
254 {"messaging", "task queue"}), None)
255 if queue_node:
256 edges.append(FlowEdge(queue_node, "worker", "consumes"))
257 else:
258 edges.append(FlowEdge(upstream, "worker", "dispatches"))
259
260 if "client" in layers or dependencies.by_category.get("http client"):
261 detail_paths = layers.get("client", [])
262 label = "External APIs"
263 if detail_paths:
264 label += f"<br/>({len(detail_paths)} adapter file(s))"
265 nodes.append(
266 FlowNode("external", label, "external-api", "stadium",
267 [Evidence("Adapter/client module", path)
268 for path in sorted(detail_paths)[:MAX_EVIDENCE_PER_NODE]])
269 )
270 edges.append(FlowEdge(upstream, "external", "calls out"))
271 signals += 1
272
273 # --- 6. response edge ------------------------------------------------
274 if endpoints:
275 edges.append(FlowEdge(upstream, "caller", "response"))
276
277 # If we learned essentially nothing, say so rather than drawing a fiction.
278 if signals <= 1 and not persisted:
279 info.note = (
280 "Not enough structural signal to infer a meaningful flow. The repository has no "
281 "recognisable layer directories, HTTP routes, or datastore dependencies — it is "
282 "most likely a library or a flat script collection."
283 )
284 info.confidence = Confidence.LOW
285 else:
286 info.confidence = Confidence.MEDIUM if signals >= 3 else Confidence.LOW
287 info.note = (
288 "Derived from directory/filename conventions plus declared dependencies. "
289 "This is the intended architecture as the layout implies it, not a verified "
290 "call graph."
291 )
292
293 info.nodes = nodes
294 info.edges = edges
295 info.mermaid = render_mermaid(nodes, edges)
296 return info

326def render_mermaid(nodes: list[FlowNode], edges: list[FlowEdge]) -> str:
327 lines = ["flowchart LR"]
328
329 for node in nodes:
330 open_wrap, close_wrap = SHAPE_WRAPPERS.get(node.shape, SHAPE_WRAPPERS["rect"])
331 lines.append(f" {node.node_id}{open_wrap}{_sanitise(node.label)}{close_wrap}")
332
333 lines.append("")
334 for edge in edges:
335 if edge.label:
336 lines.append(f" {edge.source} -->|{_sanitise(edge.label)}| {edge.target}")
337 else:
338 lines.append(f" {edge.source} --> {edge.target}")
339
340 styled = [node for node in nodes if node.layer in LAYER_STYLES]
341 if styled:
342 lines.append("")
343 for node in styled:
344 lines.append(f" style {node.node_id} {LAYER_STYLES[node.layer]}")
345
346 return "\n".join(lines)
