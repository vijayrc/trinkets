91# --- API route detection -------------------------------------------------
92HTTP_VERBS = ("get", "post", "put", "patch", "delete", "head", "options")
93
94ROUTE_PATTERNS: tuple[tuple[str, re.Pattern[str], str], ...] = (
95 # @app.get("/x") / @router.post("/x") — FastAPI, Flask 2.x, Sanic
96 ("Python decorator", re.compile(
97 r"@\w+\.(" + "|".join(HTTP_VERBS) + r")\(\s*[\"']([^\"']+)[\"']", re.IGNORECASE),
98 "python"),
99 # @app.route("/x", methods=["POST"])
100 ("Flask route", re.compile(
101 r"@\w+\.route\(\s*[\"']([^\"']+)[\"'](?:[^)]*methods\s*=\s*\[([^\]]*)\])?"), "flask"),
102 # Django urls.py
103 ("Django URLconf", re.compile(r"\b(?:path|re_path|url)\(\s*[r]?[\"']([^\"']*)[\"']"), "django"),
104 # Express / Koa / Fastify
105 ("Express", re.compile(
106 r"\b(?:app|router|server|api)\.(" + "|".join(HTTP_VERBS) + r")\(\s*[\"'`]([^\"'`]+)"),
107 "express"),
108 # Spring
109 ("Spring", re.compile(
110 r"@(Get|Post|Put|Patch|Delete|Request)Mapping\(\s*(?:value\s*=\s*)?[\"']([^\"']*)[\"']"),
111 "spring"),
112 # Go: gin / echo / chi / gorilla
113 ("Go router", re.compile(
114 r"\b\w+\.(GET|POST|PUT|PATCH|DELETE|HandleFunc|Handle)\(\s*[\"`]([^\"`]+)"), "go"),
115 # Rails routes.rb
116 ("Rails routes", re.compile(
117 r"^\s*(" + "|".join(HTTP_VERBS) + r")\s+[\"']([^\"']+)[\"']", re.MULTILINE), "rails"),
118)

121def _count_python(source: SourceFile) -> tuple[int, int]:
122 """Exact class/function counts via AST, falling back to regex on syntax errors."""
123 if not source.text:
124 return 0, 0
125 try:
126 tree = ast.parse(source.text)
127 except (SyntaxError, ValueError, RecursionError):
128 classes = len(re.findall(r"^\s*class\s+\w+", source.text, re.MULTILINE))
129 functions = len(re.findall(r"^\s*(?:async\s+)?def\s+\w+", source.text, re.MULTILINE))
130 return classes, functions
131
132 classes = functions = 0
133 for node in ast.walk(tree):
134 if isinstance(node, ast.ClassDef):
135 classes += 1
136 elif isinstance(node, ast.FunctionDef | ast.AsyncFunctionDef):
137 functions += 1
138 return classes, functions

141def _blank_comments(text: str, language: str | None) -> str:
142 """Replace comment bodies with spaces, preserving offsets and line numbers.
143
144 Without this, a commented-out or documented route (``# @app.get("/x")``)
145 counts as a real endpoint. Offsets are preserved so reported line numbers
146 still point at the right place in the original file.
147 """
148 if not language:
149 return text
150 line_prefixes, block = lang_table.COMMENT_SYNTAX.get(language, ((), None))
151 if not line_prefixes and not block:
152 return text
153
154 out: list[str] = []
155 in_block = False
156 block_start, block_end = block if block else ("", "")
157
158 for line in text.split("\n"):
159 stripped = line.lstrip()
160 indent = len(line) - len(stripped)
161
162 if in_block:
163 out.append(" " * len(line))
164 if block_end and block_end in stripped:
165 in_block = False
166 continue
167
168 if block_start and stripped.startswith(block_start):
169 out.append(" " * len(line))
170 rest = stripped[len(block_start):]
171 if not (block_end and block_end in rest):
172 in_block = True
173 continue
174
175 cut = None
176 for prefix in line_prefixes:
177 position = line.find(prefix, indent)
178 # Only treat it as a comment if it is not inside an obvious string.
179 if position != -1 and line.count('"', 0, position) % 2 == 0 \
180 and line.count("'", 0, position) % 2 == 0:
181 cut = position if cut is None else min(cut, position)
182 out.append(line if cut is None else line[:cut] + " " * (len(line) - cut))
183
184 return "\n".join(out)

187def _detect_endpoints(source: SourceFile) -> list[ApiEndpoint]:
188 if not source.text or not source.is_code:
189 return []
190
191 language = source.language or ""
192 text = _blank_comments(source.text, source.language)
193 endpoints: list[ApiEndpoint] = []
194
195 relevant: list[tuple[str, re.Pattern[str], str]] = []
196 for name, pattern, dialect in ROUTE_PATTERNS:
197 if dialect in {"python", "flask", "django"} and language != "Python":
198 continue
199 if dialect == "express" and language not in {"JavaScript", "TypeScript"}:
200 continue
201 if dialect == "spring" and language not in {"Java", "Kotlin", "Groovy"}:
202 continue
203 if dialect == "go" and language != "Go":
204 continue
205 if dialect == "rails" and language != "Ruby":
206 continue
207 relevant.append((name, pattern, dialect))
208
209 for name, pattern, dialect in relevant:
210 for match in pattern.finditer(text):
211 groups = match.groups()
212 if dialect == "flask":
213 route = groups[0]
214 methods = groups[1] if len(groups) > 1 and groups[1] else "GET"
215 method = ",".join(
216 sorted({m.strip().strip("\"'").upper() for m in methods.split(",") if m.strip()})
217 ) or "GET"
218 elif dialect == "django":
219 route, method = groups[0], "ANY"
220 elif dialect == "spring":
221 verb = groups[0]
222 method = "ANY" if verb == "Request" else verb.upper()
223 route = groups[1]
224 else:
225 method = groups[0].upper()
226 route = groups[1]
227 if method in {"HANDLEFUNC", "HANDLE"}:
228 method = "ANY"
229
230 line_number = text.count("\n", 0, match.start()) + 1
231 endpoints.append(
232 ApiEndpoint(
233 method=method,
234 route=route,
235 path=source.posix_relpath,
236 line=line_number,
237 framework=name,
238 )
239 )
240 return endpoints

243def analyse(files: list[SourceFile], skipped: int) -> tuple[list[LanguageStat], CodeStats]:
244 stats = CodeStats()
245 stats.total_files = len(files) + skipped
246 stats.analysed_files = len(files)
247 stats.skipped_files = skipped
248
249 by_language: dict[str, LanguageStat] = {}
250 counts_by_language: dict[str, dict[str, int]] = defaultdict(lambda: {"classes": 0,
251 "functions": 0})
252 exact_used: set[str] = set()
253 heuristic_used: set[str] = set()
254 file_sizes: list[tuple[str, int]] = []
255 endpoints: list[ApiEndpoint] = []
256
257 for source in files:
258 language = source.language or "Unknown"
259 entry = by_language.get(language)
260 if entry is None:
261 entry = LanguageStat(language=language, is_code=lang_table.is_code(language))
262 by_language[language] = entry
263
264 entry.files += 1
265 entry.total_lines += source.total_lines
266 entry.code_lines += source.code_lines
267 entry.comment_lines += source.comment_lines
268 entry.blank_lines += source.blank_lines
269 entry.bytes += source.size_bytes
270
271 stats.total_lines += source.total_lines
272 stats.code_lines += source.code_lines
273 stats.comment_lines += source.comment_lines
274 stats.blank_lines += source.blank_lines
275 stats.bytes += source.size_bytes
276
277 if source.is_code:
278 file_sizes.append((source.posix_relpath, source.total_lines))
279
280 if source.language in EXACT_LANGUAGES:
281 classes, functions = _count_python(source)
282 exact_used.add(source.language)
283 elif source.language in DECLARATION_PATTERNS and source.text:
284 class_pattern, function_pattern = DECLARATION_PATTERNS[source.language]
285 classes = len(class_pattern.findall(source.text)) if class_pattern else 0
286 functions = len(function_pattern.findall(source.text)) if function_pattern else 0
287 heuristic_used.add(source.language)
288 else:
289 classes = functions = 0
290
291 stats.classes += classes
292 stats.functions += functions
293 if classes or functions:
294 counts_by_language[language]["classes"] += classes
295 counts_by_language[language]["functions"] += functions
296
297 endpoints.extend(_detect_endpoints(source))
298
299 # De-duplicate endpoints that several patterns matched at the same spot.
300 unique: dict[tuple[str, int], ApiEndpoint] = {}
301 for endpoint in endpoints:
302 unique.setdefault((endpoint.path, endpoint.line), endpoint)
303 stats.endpoints = sorted(unique.values(), key=lambda e: (e.path, e.line))
304 stats.api_endpoints = len(stats.endpoints)
305
306 stats.counts_by_language = dict(counts_by_language)
307 stats.exact_parse_languages = sorted(exact_used)
308 stats.heuristic_parse_languages = sorted(heuristic_used)
309 stats.largest_files = sorted(file_sizes, key=lambda item: item[1], reverse=True)[:10]
310
311 ranked = sorted(
312 by_language.values(),
313 key=lambda entry: (entry.is_code, entry.code_lines, entry.files),
314 reverse=True,
315 )
316 return ranked, stats
