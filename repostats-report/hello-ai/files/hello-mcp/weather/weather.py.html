15logger = logging.getLogger(__name__)

17VALID_STATE_CODE = re.compile(r"^[A-Z]{2}$")

19# Shared HTTP client — reused across requests for connection pooling
20_http_client: httpx.AsyncClient | None = None
21

23def get_http_client() -> httpx.AsyncClient:
24 global _http_client
25 if _http_client is None:
26 _http_client = httpx.AsyncClient(
27 headers={"User-Agent": USER_AGENT, "Accept": "application/geo+json"},
28 timeout=30.0,
29 )
30 return _http_client

33async def make_nws_request(url: str) -> dict[str, Any] | None:
34 """Make a request to the NWS API with proper error handling."""
35 client = get_http_client()
36 try:
37 response = await client.get(url)
38 response.raise_for_status()
39 return response.json()
40 except httpx.TimeoutException:
41 logger.warning("NWS API request timed out: %s", url)
42 return None
43 except httpx.HTTPStatusError as e:
44 logger.warning("NWS API returned HTTP %s for %s", e.response.status_code, url)
45 return None
46 except Exception:
47 logger.exception("Unexpected error fetching NWS data from %s", url)
48 return None

51def format_alert(feature: dict) -> str:
52 """Format an alert feature into a readable string."""
53 props = feature.get("properties", {})
54 return f"""
55Event: {props.get("event", "Unknown")}
56Area: {props.get("areaDesc", "Unknown")}
57Severity: {props.get("severity", "Unknown")}
58Description: {props.get("description", "No description available")}
59Instructions: {props.get("instruction", "No specific instructions provided")}
60"""

63@mcp.tool()
64async def get_alerts(state: str) -> str:
65 """Get weather alerts for a US state.
66
67 Args:
68 state: Two-letter US state code (e.g. CA, NY)
69 """
70 state = state.strip().upper()
71 if not VALID_STATE_CODE.match(state):
72 return "Invalid state code. Please provide a two-letter US state code (e.g. CA, NY)."
73
74 url = f"{NWS_API_BASE}/alerts/active/area/{state}"
75 data = await make_nws_request(url)
76
77 if not data or "features" not in data:
78 return "Unable to fetch alerts or no alerts found."
79
80 if not data["features"]:
81 return "No active alerts for this state."
82
83 alerts = [format_alert(feature) for feature in data["features"]]
84 return "\n---\n".join(alerts)

87@mcp.tool()
88async def get_forecast(latitude: float, longitude: float) -> str:
89 """Get weather forecast for a location.
90
91 Args:
92 latitude: Latitude of the location
93 longitude: Longitude of the location
94 """
95 if not (-90 <= latitude <= 90) or not (-180 <= longitude <= 180):
96 return "Invalid coordinates. Latitude must be -90 to 90, longitude -180 to 180."
97
98 # First get the forecast grid endpoint
99 points_url = f"{NWS_API_BASE}/points/{latitude},{longitude}"
100 points_data = await make_nws_request(points_url)
101
102 if not points_data:
103 return "Unable to fetch forecast data for this location."
104
105 # Get the forecast URL from the points response
106 forecast_url = points_data.get("properties", {}).get("forecast")
107 if not forecast_url:
108 return "Location is not supported by the NWS API (only US locations are covered)."
109
110 forecast_data = await make_nws_request(forecast_url)
111
112 if not forecast_data:
113 return "Unable to fetch detailed forecast."
114
115 # Format the periods into a readable forecast
116 periods = forecast_data["properties"]["periods"]
117 forecasts = []
118 for period in periods[:5]: # Only show next 5 periods
119 forecast = f"""
120{period["name"]}:
121Temperature: {period["temperature"]}°{period["temperatureUnit"]}
122Wind: {period["windSpeed"]} {period["windDirection"]}
123Forecast: {period["detailedForecast"]}
124"""
125 forecasts.append(forecast)
126
127 return "\n---\n".join(forecasts)
