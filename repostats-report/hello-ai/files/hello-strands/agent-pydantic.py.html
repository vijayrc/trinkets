1from pydantic import BaseModel, Field
2from strands import Agent
3from strands_tools import calculator
4from typing import Optional
5
6# Example 1
7# 1.1) Define the Pydantic model
8class PersonInfo(BaseModel):
9 """Model that contains information about a Person"""
10 name: str = Field(description="Name of the person")
11 age: int = Field(description="Age of the person")
12 occupation: str = Field(description="Occupation of the person")
13
14# 1.2) Pass the model to the agent
15agent = Agent()
16result = agent(
17 "John Smith is a 30 year-old software engineer",
18 structured_output_model=PersonInfo
19)
20
21# 3) Access the `structured_output` from the result
22person_info: PersonInfo = result.structured_output
23print(f"Name: {person_info.name}") # "John Smith"
24print(f"Age: {person_info.age}") # 30
25print(f"Job: {person_info.occupation}") # "software engineer"

27# Example 2
28class MathResult(BaseModel):
29 operation: str = Field(description="the performed operation")
30 result: int = Field(description="the result of the operation")
31
32tool_agent = Agent(
33 tools=[calculator]
34)
35res = tool_agent("What is 42 + 8", structured_output_model=MathResult)
36math_info: MathResult = res.structured_output
37print(f"operation: {math_info.operation}")
38print(f"result: {math_info.result}")
