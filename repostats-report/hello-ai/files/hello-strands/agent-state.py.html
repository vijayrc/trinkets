27# 3 STATE WITH TOOLS -----------------------------------------------------------
28@tool(context=True)
29def track_user_action(action: str, tool_context: ToolContext):
30 """Track user actions in agent state.
31
32 Args:
33 action: The action to track
34 """
35 action_count = tool_context.agent.state.get("action_count") or 0
36 tool_context.agent.state.set("action_count", action_count + 1)
37 tool_context.agent.state.set("last_action", action)
38 return f"Action '{action}' recorded. Total actions: {action_count + 1}"
39
40@tool(context=True)
41def get_user_stats(tool_context: ToolContext):
42 """Get user statistics from agent state."""
43 action_count = tool_context.agent.state.get("action_count") or 0
44 last_action = tool_context.agent.state.get("last_action") or "none"
45 return f"Actions performed: {action_count}, Last action: {last_action}"
46
47agent3 = Agent(tools=[track_user_action, get_user_stats])
48agent3("Track that I logged in")
49agent3("Track that I viewed my profile")
50print(f"Actions taken: {agent3.state.get('action_count')}", end="\n")
51print(f"Last action: {agent3.state.get('last_action')}", end="\n")

54def custom_callback_handler(**kwargs):

60 state["counter"] = 0
