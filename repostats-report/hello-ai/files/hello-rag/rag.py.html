13class RAGSystem:

18 def _initialize_qa_chain(self):
19 # Load the document
20 loader = TextLoader('data/my_document.txt')
21 documents = loader.load()
22
23 # Split the document into chunks
24 splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
25 document_chunks = splitter.split_documents(documents)
26
27 # Initialize embeddings with OpenAI API key
28 embeddings = OpenAIEmbeddings(openai_api_key=self.openai_api_key)
29
30 # FAISS expects document objects and the embedding model
31 vector_store = FAISS.from_documents(document_chunks, embeddings)
32
33 # Use the vector store's retriever
34 retriever = vector_store.as_retriever()
35
36 # Initialize the LLM (using OpenAI)
37 llm = OpenAI(openai_api_key=self.openai_api_key)
38
39 # Set up the retrieval-based QA chain
40 return RetrievalQA.from_chain_type(
41 llm=llm,
42 chain_type="stuff",
43 retriever=retriever
44 )
45
46 async def get_response(self, query: str) -> str:
