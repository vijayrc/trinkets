1import logging
2from strands import Agent
3from strands.multiagent import Swarm
4
5# Enable debug logs and print them to stderr
6logging.getLogger("strands.multiagent").setLevel(logging.DEBUG)
7logging.basicConfig(
8 format="%(levelname)s | %(name)s | %(message)s",
9 handlers=[logging.StreamHandler()]
10)
11
12# Create specialized agents
13researcher = Agent(name="researcher", system_prompt="You are a research specialist...")
14coder = Agent(name="coder", system_prompt="You are a coding specialist...")
15reviewer = Agent(name="reviewer", system_prompt="You are a code review specialist...")
16architect = Agent(name="architect", system_prompt="You are a system architecture specialist...")
17
18# Create a swarm with these agents, starting with the researcher
19swarm = Swarm(
20 [coder, researcher, reviewer, architect],
21 entry_point=researcher, # Start with the researcher
22 max_handoffs=20,
23 max_iterations=20,
24 execution_timeout=900.0, # 15 minutes
25 node_timeout=300.0, # 5 minutes per agent
26 repetitive_handoff_detection_window=8, # There must be >= 3 unique agents in the last 8 handoffs
27 repetitive_handoff_min_unique_agents=3
28)
29
30# Execute the swarm on a task
31result = swarm("Design and implement a simple REST API for a todo app into /Users/vijayarajchakravarthy/work/tenet/apps/")
32# Or use invoke_async for async execution: result = await swarm.invoke_async(...)
33
34# Access the final result
35print(f"Status: {result.status}")
36print(f"Node history: {[node.node_id for node in result.node_history]}")
