3import requests

17class LedgerPlugin(Plugin):
18 """Plugin that exposes Mambu General Ledger API operations as agent tools."""
19
20 name = "ledger-plugin"
21 log = logging.getLogger("LedgerPlugin")
22
23 def init_agent(self, agent: "Agent") -> None:
24 self.tenant = agent.state.get("tenant")
25 self.user = agent.state.get("user")
26 self.password = agent.state.get("password")
27 self.log.info("agent-init: tenant=%s user=%s", self.tenant, self.user)
28
29 # --- hooks ---
30
31 @hook
32 def do_before_tool(self, event: BeforeToolCallEvent) -> None:
33 self.log.info("tool-start: %s | input: %s", event.tool_use['name'], event.tool_use['input'])
34
35 @hook
36 def do_after_tool(self, event: AfterToolCallEvent) -> None:
37 self.log.info("tool-end: %s", event.tool_use['name'])
38
39 # --- private helpers ---
40
41 def _url(self, path: str) -> str:
42 return f"https://{self.tenant}.mambu.com/api/{path}"
43
44 def _headers(self) -> dict:
45 return {
46 'Accept': 'application/vnd.mambu.v2+json',
47 'Content-Type': 'application/json',
48 'Idempotency-Key': str(uuid.uuid4())
49 }
50
51 def _auth(self) -> HTTPBasicAuth:
52 return HTTPBasicAuth(self.user, self.password)
53
54 def _get(self, path: str) -> dict:
55 url = self._url(path)
56 self.log.info("GET %s", url)
57 response = requests.get(url, auth=self._auth(), headers=self._headers())
58 self.log.info("response: code=%s body=%s", response.status_code, response.json())
59 return response.json()
60
61 def _post(self, path: str, payload: dict) -> dict:
62 url = self._url(path)
63 self.log.info("POST %s payload=%s", url, payload)
64 response = requests.post(url, auth=self._auth(), headers=self._headers(), json=payload)
65 self.log.info("response: code=%s body=%s", response.status_code, response.json())
66 return response.json()

68 # --- tools ---
69
70 @tool
71 def call_get_all_gl_accounts(self) -> dict:
72 """Fetch all general ledger accounts.
73 API: https://docs.mambu.com/api/api-v2/glaccounts/get-all/
74 """
75 return self._get("glaccounts")
76
77 @tool
78 def call_get_gl_account(self, gl_account_id: str) -> dict:
79 """Fetch a general ledger account by ID.
80 API: https://docs.mambu.com/api/api-v2/glaccounts/get-by-id/
81
82 Args:
83 gl_account_id: The general ledger account ID to fetch
84 """
85 return self._get(f"glaccounts/{gl_account_id}")
86
87 @tool
88 def call_create_gl_account(self, name: str = "", account_type: str = "ASSET", currency_code: str = "USD") -> dict:
89 """Create a new general ledger account.
90 API: https://docs.mambu.com/api/api-v2/glaccounts/create/
91
92 Args:
93 name: The name of the GL account (generated if not provided)
94 account_type: Type of GL account — ASSET, LIABILITY, EQUITY, INCOME, or EXPENSE
95 currency_code: ISO 4217 currency code (e.g. USD, EUR, GBP)
96 """
97 name = default_label(name, "tenet_gl_account")
98 payload = {
99 "name": name,
100 "type": account_type,
101 "currency": {
102 "code": currency_code,
103 "currencyCode": currency_code
104 },
105 "description": "tenet-test ledger account",
106 "glCode": "3200",
107 "usage": "DETAIL",
108 "stripTrailingZeros": True,
109 "allowManualJournalEntries": True
110 }
111 return self._post("glaccounts", payload)
