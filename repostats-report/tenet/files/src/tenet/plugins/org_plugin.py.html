17class OrgPlugin(Plugin):
18 """Plugin that exposes Mambu Organisation API operations as agent tools."""
19
20 name = "org-plugin"
21 log = logging.getLogger("OrgPlugin")
22
23 def init_agent(self, agent: "Agent") -> None:
24 self.tenant = agent.state.get("tenant")
25 self.user = agent.state.get("user")
26 self.password = agent.state.get("password")
27 self.log.info("agent-init: tenant=%s user=%s", self.tenant, self.user)

32 def do_before_tool(self, event: BeforeToolCallEvent) -> None:
33 self.log.info("tool-start: %s | input: %s", event.tool_use['name'], event.tool_use['input'])
34

36 def do_after_tool(self, event: AfterToolCallEvent) -> None:

39 # --- private helpers ---
40
41 def _url(self, path: str) -> str:
42 return f"https://{self.tenant}.mambu.com/api/{path}"
43
44 def _headers(self) -> dict:
45 return {
46 'Accept': 'application/vnd.mambu.v2+json',
47 'Content-Type': 'application/json',
48 }
49
50 def _auth(self) -> HTTPBasicAuth:
51 return HTTPBasicAuth(self.user, self.password)
52
53 def _get(self, path: str) -> dict:
54 url = self._url(path)
55 self.log.info("GET %s", url)
56 response = requests.get(url, auth=self._auth(), headers=self._headers())
57 self.log.info("response: code=%s body=%s", response.status_code, response.json())
58 return response.json()
59
60 def _post(self, path: str, payload: dict) -> dict:
61 url = self._url(path)
62 self.log.info("POST %s payload=%s", url, payload)
63 response = requests.post(url, auth=self._auth(), headers=self._headers(), json=payload)
64 self.log.info("response: code=%s body=%s", response.status_code, response.json())
65 return response.json()
66
67 def _put(self, path: str, payload: dict) -> dict:
68 url = self._url(path)
69 self.log.info("PUT %s payload=%s", url, payload)
70 response = requests.put(url, auth=self._auth(), headers=self._headers(), json=payload)
71 self.log.info("response: code=%s body=%s", response.status_code, response.json())
72 return response.json()
